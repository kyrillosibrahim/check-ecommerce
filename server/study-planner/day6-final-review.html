GROUP BY + HAVING + Aggregate Functions

SQL - GROUP BY
SELECT department_id,
       COUNT(*) AS emp_count,
       AVG(salary) AS avg_sal,
       MAX(salary) AS max_sal,
       MIN(salary) AS min_sal,
       SUM(salary) AS total_sal
FROM employees
GROUP BY department_id
HAVING COUNT(*) > 5
ORDER BY avg_sal DESC;

Subqueries - الاستعلامات الفرعية

SQL - Subqueries
-- IN: هل القيمة موجودة في القائمة؟
SELECT * FROM employees
WHERE department_id IN (
    SELECT department_id FROM departments WHERE location_id = 1700
);

-- EXISTS: هل في نتائج؟
SELECT * FROM departments d
WHERE EXISTS (
    SELECT 1 FROM employees e WHERE e.department_id = d.department_id
);

-- ANY / ALL
SELECT * FROM employees
WHERE salary > ALL (SELECT salary FROM employees WHERE dept_id = 30);

ملخص PL/SQL

الهيكل الأساسي - Block Structure

PL/SQL - Basic Block
DECLARE
    v_name  VARCHAR2(50);
    v_sal   NUMBER(8,2) := 0;
    v_date  DATE := SYSDATE;
BEGIN
    SELECT first_name, salary
    INTO v_name, v_sal
    FROM employees
    WHERE employee_id = 100;

    DBMS_OUTPUT.PUT_LINE(v_name || ': ' || v_sal);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Not found!');
END;
/

    v_emp  employees%ROWTYPE;
BEGIN

END;

-- WHILE LOOP

Cursors - المؤشرات

PL/SQL - Cursors
-- Implicit Cursor (SELECT INTO)
SELECT salary INTO v_sal FROM employees WHERE employee_id = 100;

-- Cursor FOR LOOP (الأسهل والأفضل)
FOR emp IN (SELECT first_name, salary FROM employees) LOOP
    DBMS_OUTPUT.PUT_LINE(emp.first_name || ': ' || emp.salary);
END LOOP;

-- Explicit Cursor
DECLARE
    CURSOR c_emp IS SELECT first_name, salary FROM employees;
    v_rec c_emp%ROWTYPE;
BEGIN
    OPEN c_emp;
    LOOP
        FETCH c_emp INTO v_rec;
        EXIT WHEN c_emp%NOTFOUND;
        DBMS_OUTPUT.PUT_LINE(v_rec.first_name);
    END LOOP;
    CLOSE c_emp;
END;
/

Exception Handling - التعامل مع الأخطاء

PL/SQL - Exceptions
DECLARE
    e_custom EXCEPTION;
BEGIN
    -- الكود
    IF condition THEN
        RAISE e_custom;
    END IF;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No data!');
    WHEN TOO_MANY_ROWS THEN
        DBMS_OUTPUT.PUT_LINE('Too many rows!');
    WHEN e_custom THEN
        DBMS_OUTPUT.PUT_LINE('Custom error!');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;
/

Triggers - المشغلات

PL/SQL - Trigger
CREATE OR REPLACE TRIGGER trg_salary_check
BEFORE INSERT OR UPDATE ON employees
FOR EACH ROW
BEGIN
    -- :NEW = القيمة الجديدة, :OLD = القيمة القديمة
    IF :NEW.salary < 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Salary must be positive!');

END;

Packages - الحزم

PL/SQL - Package Spec
-- Package Specification (الواجهة العامة)
CREATE OR REPLACE PACKAGE emp_pkg IS
    PROCEDURE hire_emp(p_name VARCHAR2, p_sal NUMBER);
    FUNCTION  get_sal(p_id NUMBER) RETURN NUMBER;
END emp_pkg;
/
PL/SQL - Package Body
-- Package Body (التنفيذ)
CREATE OR REPLACE PACKAGE BODY emp_pkg IS
    PROCEDURE hire_emp(p_name VARCHAR2, p_sal NUMBER) IS
    BEGIN
        INSERT INTO employees(first_name, salary)
        VALUES(p_name, p_sal);
    END;

    FUNCTION get_sal(p_id NUMBER) RETURN NUMBER IS
        v_sal NUMBER;
    BEGIN
        SELECT salary INTO v_sal FROM employees WHERE employee_id = p_id;
        RETURN v_sal;
    END;
END emp_pkg;

VLOOKUP - البحث العمودي

Excel - VLOOKUP
=VLOOKUP(lookup_value, table_array, col_index_num, [range_lookup])

// مثال: البحث عن اسم موظف بالـ ID
=VLOOKUP(A2, Sheet2!A:D, 2, FALSE)

// FALSE = تطابق تام (Exact Match)
// TRUE  = تطابق تقريبي (Approximate Match)
قواعد VLOOKUP المهمة: قيمة البحث لازم تكون في أول عمود من الـ table_array. بيبحث من اليسار لليمين فقط. لو عايز تبحث لليسار استخدم INDEX+MATCH.

HLOOKUP - البحث الأفقي

Excel - HLOOKUP
=HLOOKUP(lookup_value, table_array, row_index_num, [range_lookup])

// مثال: البحث في جدول أفقي
=HLOOKUP("Q1", A1:D3, 2, FALSE)

XLOOKUP - البديل الحديث

Excel - XLOOKUP
=XLOOKUP(lookup_value, lookup_array, return_array, [if_not_found], [match_mode], [search_mode])

// مثال: بحث مع رسالة لو مش لاقي
=XLOOKUP(A2, IDs, Names, "Not Found")

// المميزات: بيبحث يمين ويسار، بيدعم if_not_found

INDEX + MATCH - الكومبو القوي

Excel - INDEX + MATCH
// INDEX: بيجيب قيمة من صف ومود محددين

// الكومبو: بحث في أي اتجاه!

مقابلات العمل

أسئلة مقابلة متوقعة

أهم 20 سؤال ممكن يتسألوا في مقابلات العمل على Oracle و Excel. كل سؤال مع إجابة مختصرة ومباشرة.

أسئلة Oracle Database (10 أسئلة)

1. ما الفرق بين WHERE و HAVING؟

WHERE بتفلتر الصفوف قبل التجميع (GROUP BY) وبتشتغل على صفوف فردية. HAVING بتفلتر بعد التجميع وبتشتغل على نتائج الدوال التجميعية (COUNT, SUM, AVG). مثلاً: HAVING COUNT(*) > 5 مش هتشتغل في WHERE.

2. ما هو JOIN؟ وما أنواعه؟

JOIN بيربط جدولين أو أكتر بناءً على شرط مشترك. الأنواع: INNER JOIN (المشترك فقط)، LEFT JOIN (كل اليسار + المشترك)، RIGHT JOIN (كل اليمين + المشترك)، FULL JOIN (كل الصفوف)، CROSS JOIN (كل التركيبات)، SELF JOIN (الجدول مع نفسه).

3. ما هو PL/SQL؟

PL/SQL (Procedural Language/SQL) هو امتداد لـ SQL من Oracle بيضيف إمكانيات البرمجة: متغيرات، شروط (IF/ELSE)، حلقات (LOOP)، التعامل مع الأخطاء (EXCEPTION)، وإنشاء Procedures و Functions و Triggers. بيشتغل على جانب السيرفر وبيحسّن الأداء.

4. ما الفرق بين Procedure و Function؟

Procedure: مش بترجع قيمة مباشرة (بتستخدم OUT parameters)، ومش بتتسخدم في SELECT. Function: بترجع قيمة واحدة بـ RETURN، وتقدر تستخدمها في SELECT. الـ Procedure للعمليات (DML)، والـ Function للحسابات.

5. ما هو Trigger؟

Trigger هو كود PL/SQL بيتنفذ تلقائياً لما يحصل حدث معين على جدول (INSERT, UPDATE, DELETE). ممكن يكون BEFORE أو AFTER الحدث، ولكل صف (FOR EACH ROW) أو مرة واحدة. بيستخدم :NEW و :OLD للوصول للقيم الجديدة والقديمة.

6. ما هو Cursor؟

Cursor هو مؤشر بيقرأ نتائج استعلام صف بصف. في نوعين: Implicit (SELECT INTO لصف واحد) و Explicit (DECLARE, OPEN, FETCH, CLOSE لصفوف متعددة). أسهل طريقة هي Cursor FOR LOOP اللي بيعمل كل الخطوات تلقائياً.

7. ما هو Index ولماذا نستخدمه؟

Index هو هيكل بيانات بيسرّع عمليات البحث (SELECT) زي فهرس الكتاب. بدل ما يقرأ كل الصفوف (Full Table Scan)، بيروح للبيانات مباشرة. لكن كتر الفهارس بيبطئ INSERT/UPDATE/DELETE لأنه لازم يحدّث الفهرس مع كل تغيير.

8. ما هو View؟

View هو استعلام محفوظ بيتعامل معاه كأنه جدول افتراضي. مش بيخزن بيانات - بس بيحفظ الـ SELECT. فوائده: تبسيط الاستعلامات المعقدة، أمان (إخفاء أعمدة حساسة)، وإعادة استخدام. ممكن يكون WITH CHECK OPTION أو READ ONLY.

9. كيف تتعامل مع الأخطاء في PL/SQL؟

باستخدام EXCEPTION block. الأخطاء الشائعة: NO_DATA_FOUND (ما لقيش بيانات)، TOO_MANY_ROWS (SELECT INTO رجع أكتر من صف)، WHEN OTHERS (أي خطأ تاني). تقدر تعمل Custom Exception بـ RAISE أو RAISE_APPLICATION_ERROR(-20001, 'msg').

10. ما هو Package؟

Package هو حاوية بتجمع Procedures و Functions و Variables و Cursors مع بعض. بيتكون من جزئين: Specification (الواجهة العامة - بيحدد إيه المتاح) و Body (التنفيذ الفعلي). الفوائد: تنظيم الكود، Encapsulation، وأداء أفضل.

أسئلة Excel (10 أسئلة)

1. ما الفرق بين VLOOKUP و INDEX+MATCH؟

VLOOKUP بيبحث في أول عمود ويرجع قيمة من عمود على اليمين فقط. INDEX+MATCH بيبحث في أي اتجاه (يمين أو يسار)، أسرع في الأداء، وأكثر مرونة. VLOOKUP أسهل في الكتابة لكن INDEX+MATCH أقوى.

2. ما هو Pivot Table؟

Pivot Table أداة تلخيص بتاخد بيانات كبيرة (آلاف الصفوف) وتلخصها في جدول صغير. بتسحب الأعمدة لأربع مناطق: Rows, Columns, Values (SUM/COUNT/AVG), Filters. تقدر تغيّر التلخيص بالسحب والإفلات بدون معادلات.

3. ما الفرق بين COUNTIF و SUMIF؟

COUNTIF بيعدّ عدد الخلايا اللي تحقق شرط معين (كم موظف مرتبه أكتر من 5000). SUMIF بيجمع قيم الخلايا اللي تحقق شرط (مجموع مرتبات قسم معين). في كمان COUNTIFS و SUMIFS لشروط متعددة.

4. ما هو Conditional Formatting؟

Conditional Formatting بيغيّر شكل الخلايا تلقائياً حسب قيمتها. مثلاً: تلوين المرتبات العالية بالأخضر والمنخفضة بالأحمر، Data Bars لتمثيل القيم بصريًا، Icon Sets لإشارات المرور. ممكن تستخدم معادلة مخصصة كمان.

5. ما هو Data Validation؟

Data Validation بيحدد إيه البيانات المسموح إدخالها في خلية. مثلاً: أرقام بين 1 و 100، قائمة منسدلة (Dropdown) بخيارات محددة، تواريخ في نطاق معين. بيمنع الأخطاء في إدخال البيانات وبيحافظ على جودة البيانات.

6. ما هو Macro؟

Macro هو تسجيل لمجموعة خطوات بتتنفذ تلقائياً. تقدر تسجل حركاتك (Record Macro) أو تكتب كود VBA يدوياً. الملف لازم يتحفظ كـ .xlsm. الماكروز بتوفر وقت كبير في المهام المتكررة زي تنسيق التقارير.

7. ما هو XLOOKUP؟

XLOOKUP هو البديل الحديث لـ VLOOKUP (متوفر في Excel 365/2021+). مميزاته: بيبحث في أي اتجاه، بيدعم رسالة مخصصة لو مش لاقي (if_not_found)، أسرع في الأداء، وأبسط في الكتابة. الصيغة: =XLOOKUP(value, lookup, return, "Not Found").

8. ما الفرق بين IF و IFS؟

IF بتتعامل مع شرط واحد (ولو عايز شروط متعددة تعمل Nested IF وده بيبقى معقد). IFS (Excel 2019+) بتتعامل مع شروط متعددة في دالة واحدة بدون تداخل. IFS أسهل في القراءة والصيانة لما يكون عندك أكتر من شرطين.

9. ما هي فوائد Named Ranges؟

Named Ranges بتسمي نطاق خلايا باسم واضح (مثلاً Salaries بدل B2:B100). الفوائد: المعادلات أسهل في القراءة والفهم، أسهل في الصيانة (تغيّر النطاق مرة واحدة)، بتشتغل عبر كل الشيتات، وبتقلل الأخطاء.

10. ما هي أدوات What-If Analysis؟

ثلاث أدوات: Goal Seek (عكس المعادلة - بتحدد النتيجة وبيلاقيلك المدخل)، Data Tables (بتعرض نتائج معادلة لقيم مختلفة في جدول)، Scenario Manager (بتحفظ سيناريوهات مختلفة وتقارن بينها - متفائل/متشائم/واقعي).

تمرين Oracle 2: JOIN + GROUP BY + HAVING

متوسط

اكتب استعلام يعرض اسم كل قسم وعدد الموظفين فيه ومتوسط المرتبات وأعلى مرتب. اعرض بس الأقسام اللي فيها أكتر من 3 موظفين ومتوسط المرتبات أكبر من 6000. رتّب بمتوسط المرتبات تنازلياً.

تلميح: هتحتاج JOIN بين employees و departments. استخدم GROUP BY على اسم القسم، و HAVING لفلترة التجميع. الدوال: COUNT, AVG, MAX.
SQL
SELECT d.department_name,
       COUNT(*) AS emp_count,
       ROUND(AVG(e.salary), 2) AS avg_salary,
       MAX(e.salary) AS max_salary
FROM employees e
JOIN departments d ON e.department_id = d.department_id
GROUP BY d.department_name
HAVING COUNT(*) > 3
   AND AVG(e.salary) > 6000
ORDER BY avg_salary DESC;

تمرين Oracle 3: Subquery + View

متوسط

تلميح: في الـ Subquery، احسب AVG(salary) لنفس القسم باستخدام WHERE department_id = e.department_id (Correlated). بعدين حط الاستعلام في CREATE VIEW.
SQL
-- 1) Correlated Subquery
SELECT e.first_name, e.salary, e.department_id
FROM employees e
WHERE e.salary > (
    SELECT AVG(e2.salary)
    FROM employees e2
    WHERE e2.department_id = e.department_id
);

-- 2) تحويل لـ View
CREATE OR REPLACE VIEW v_above_avg_salary AS
SELECT e.first_name, e.salary, e.department_id
FROM employees e
WHERE e.salary > (
    SELECT AVG(e2.salary)
    FROM employees e2
    WHERE e2.department_id = e.department_id
);

تمرين Oracle 4: PL/SQL Block + Cursor + Exception

صعب

اكتب PL/SQL block يقرأ كل الموظفين في قسم 30 باستخدام Cursor FOR LOOP. لكل موظف: لو مرتبه أقل من 5000 اطبع "Low Salary" مع اسمه، لو بين 5000 و 10000 اطبع "Medium"، لو أكتر اطبع "High". في الآخر اطبع عدد الموظفين الكلي. تعامل مع الأخطاء.

تلميح: استخدم Cursor FOR LOOP مع SELECT من employees WHERE department_id = 30. جوا الـ LOOP استخدم IF/ELSIF. عرّف counter وزوّده كل لفة. حط EXCEPTION WHEN OTHERS في الآخر.
PL/SQL
DECLARE
    v_count NUMBER := 0;
    v_level VARCHAR2(20);
BEGIN
    FOR emp IN (
        SELECT first_name, salary
        FROM employees
        WHERE department_id = 30
    ) LOOP
        v_count := v_count + 1;

        IF emp.salary < 5000 THEN
            v_level := 'Low Salary';
        ELSIF emp.salary <= 10000 THEN
            v_level := 'Medium';
        ELSE
            v_level := 'High';
        END IF;

        DBMS_OUTPUT.PUT_LINE(emp.first_name || ' - ' || emp.salary || ' - ' || v_level);
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Total employees: ' || v_count);
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;
/

تمرين Oracle 5: Procedure + Function

صعب

1) اكتب Function اسمها get_annual_salary تاخد employee_id وترجع المرتب السنوي (salary * 12 + salary * NVL(commission_pct, 0) * 12). 2) اكتب Procedure اسمها transfer_employee تنقل موظف من قسم لقسم (تاخد emp_id و new_dept_id). لو الموظف مش موجود، ارفع خطأ مخصص.

تلميح: الـ Function تستخدم SELECT INTO لجلب salary و commission_pct. الـ Procedure تستخدم UPDATE SET department_id = new_dept_id WHERE employee_id = emp_id. تحقق بـ SQL%ROWCOUNT لو الموظف موجود.
PL/SQL - Function
CREATE OR REPLACE FUNCTION get_annual_salary(
    p_emp_id IN NUMBER
) RETURN NUMBER
IS
    v_salary NUMBER;
    v_comm   NUMBER;
BEGIN
    SELECT salary, NVL(commission_pct, 0)
    INTO v_salary, v_comm
    FROM employees
    WHERE employee_id = p_emp_id;

    RETURN (v_salary + v_salary * v_comm) * 12;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN -1;
END;
/
PL/SQL - Procedure
CREATE OR REPLACE PROCEDURE transfer_employee(
    p_emp_id   IN NUMBER,
    p_new_dept IN NUMBER
) IS
BEGIN
    UPDATE employees
    SET department_id = p_new_dept
    WHERE employee_id = p_emp_id;

    IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'Employee not found: ' || p_emp_id);
    END IF;

    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Transfer done!');
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;
/

تمرين Oracle 6: Trigger + Package (تحدي)

تحدي

1) اكتب Trigger اسمه trg_salary_audit: قبل أي UPDATE على salary في جدول employees، يسجّل في جدول salary_log (emp_id, old_salary, new_salary, change_date, changed_by). 2) اكتب Package اسمه emp_reports فيه: Function تحسب متوسط مرتبات قسم، و Procedure تطبع تقرير بكل الموظفين في قسم معين مع تصنيفهم.

تلميح: الـ Trigger يكون BEFORE UPDATE OF salary ON employees FOR EACH ROW. استخدم :OLD.salary و :NEW.salary. الـ Package Spec يعلن عن الـ Function و Procedure، والـ Body يحتوي التنفيذ.
SQL - Create Log Table
CREATE TABLE salary_log (
    log_id       NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    emp_id       NUMBER(6),
    old_salary   NUMBER(8,2),
    new_salary   NUMBER(8,2),
    change_date  DATE DEFAULT SYSDATE,
    changed_by   VARCHAR2(50) DEFAULT USER
);
PL/SQL - Trigger
CREATE OR REPLACE TRIGGER trg_salary_audit
BEFORE UPDATE OF salary ON employees
FOR EACH ROW
BEGIN
    INSERT INTO salary_log (emp_id, old_salary, new_salary)
    VALUES (:OLD.employee_id, :OLD.salary, :NEW.salary);
END;
/
PL/SQL - Package
-- Package Specification
CREATE OR REPLACE PACKAGE emp_reports IS
    FUNCTION  get_dept_avg(p_dept_id NUMBER) RETURN NUMBER;
    PROCEDURE print_dept_report(p_dept_id NUMBER);
END emp_reports;
/

-- Package Body
CREATE OR REPLACE PACKAGE BODY emp_reports IS

    FUNCTION get_dept_avg(p_dept_id NUMBER) RETURN NUMBER IS
        v_avg NUMBER;
    BEGIN
        SELECT AVG(salary) INTO v_avg
        FROM employees WHERE department_id = p_dept_id;
        RETURN ROUND(v_avg, 2);
    END;

    PROCEDURE print_dept_report(p_dept_id NUMBER) IS
    BEGIN
        DBMS_OUTPUT.PUT_LINE('=== Department ' || p_dept_id || ' Report ===');
        DBMS_OUTPUT.PUT_LINE('Avg Salary: ' || get_dept_avg(p_dept_id));
        FOR emp IN (
            SELECT first_name, salary FROM employees
            WHERE department_id = p_dept_id ORDER BY salary DESC
        ) LOOP
            DBMS_OUTPUT.PUT_LINE(
                emp.first_name || ' - ' || emp.salary || ' - ' ||
                CASE WHEN emp.salary > 10000 THEN 'Senior'
                     WHEN emp.salary > 5000  THEN 'Mid'
                     ELSE 'Junior' END
            );
        END LOOP;
    END;

END emp_reports;
/

=SUMIFS(C2:C100, B2:B100, "IT", D2:D100, ">2020")

=IFS(C2>10000, "Senior", C2>5000, "Mid", C2>3000, "Junior", TRUE, "Trainee")

الحل التفصيلي:

  1. Data Validation - القسم (B2): Data > Validation > Allow: List > Source: Sales,IT,HR,Finance
  2. Data Validation - الميزانية (B3): Data > Validation > Allow: Whole Number > Between 1000 and 100000
  3. معادلة الضريبة (B5): =B3*15% أو =B3*0.15
  4. Goal Seek: Data > What-If Analysis > Goal Seek > Set cell: B5 > To value: 7500 > By changing cell: B3 > النتيجة: B3 = 50000
  5. Data Table: اكتب القيم (10000-50000) في D3:D7. في E2 حط =B5. حدد D2:E7 > Data > What-If > Data Table > Column input cell: B3
  6. Macro: View > Macros > Record Macro > اسم: FormatBudget. نفّذ: حدد A1:A5 واعمل Bold. حدد B2:B5 واختار خلفية صفراء. حدد A:B واعمل AutoFit. وقّف التسجيل.
VBA - Macro Code
Sub FormatBudget()
    ' Bold للعناوين
    Range("A1:A5").Font.Bold = True

    ' خلفية صفراء للقيم
    Range("B2:B5").Interior.Color = RGB(255, 255, 200)

    ' AutoFit
    Columns("A:B").AutoFit

    MsgBox "Budget formatted successfully!"
End Sub
