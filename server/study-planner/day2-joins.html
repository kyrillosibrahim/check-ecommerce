JOINs - ربط الجداول ببعض

البيانات التجريبية

هنشتغل على 3 جداول. لو معملتهمش قبل كده:

الجداول الـ 3
-- جدول الأقسام (4 أقسام) CREATE TABLE departments ( dept_id NUMBER(4) PRIMARY KEY, dept_name VARCHAR2(50) NOT NULL, location VARCHAR2(50) ); INSERT INTO departments VALUES (10, 'المبيعات', 'القاهرة'); INSERT INTO departments VALUES (20, 'التقنية', 'الإسكندرية'); INSERT INTO departments VALUES (30, 'الموارد البشرية', 'القاهرة'); INSERT INTO departments VALUES (40, 'التسويق', 'المنصورة'); -- جدول الموظفين (8 موظفين) CREATE TABLE employees ( emp_id NUMBER(6) PRIMARY KEY, first_name VARCHAR2(30), last_name VARCHAR2(30), salary NUMBER(10,2), hire_date DATE, dept_id NUMBER(4) REFERENCES departments(dept_id), manager_id NUMBER(6) ); INSERT INTO employees VALUES(101,'أحمد','محمد',8000,DATE '2023-01-15',10,NULL); INSERT INTO employees VALUES(102,'سارة','علي',12000,DATE '2022-06-01',20,101); INSERT INTO employees VALUES(103,'محمد','حسن',6500,DATE '2023-09-20',10,101); INSERT INTO employees VALUES(104,'فاطمة','أحمد',15000,DATE '2021-03-10',20,101); INSERT INTO employees VALUES(105,'عمر','خالد',7000,DATE '2024-01-05',30,102); INSERT INTO employees VALUES(106,'نور','سعيد',9500,DATE '2022-11-15',20,104); INSERT INTO employees VALUES(107,'يوسف','إبراهيم',5500,DATE '2024-03-01',10,103); INSERT INTO employees VALUES(108,'منى','عبدالله',11000,DATE '2023-07-20',NULL,101); -- جدول المشاريع CREATE TABLE projects ( project_id NUMBER(4) PRIMARY KEY, project_name VARCHAR2(60), dept_id NUMBER(4) REFERENCES departments(dept_id), budget NUMBER(12,2) ); INSERT INTO projects VALUES(1,'موقع التجارة الإلكترونية',20,500000); INSERT INTO projects VALUES(2,'حملة التسويق الرقمي',40,200000); INSERT INTO projects VALUES(3,'نظام إدارة المبيعات',10,350000); INSERT INTO projects VALUES(4,'تطبيق الموبايل',20,750000); COMMIT;

ليه محتاجين JOIN؟

في الـ Relational Database، البيانات متقسمة على جداول مختلفة عشان نتجنب التكرار.

جدول الموظفين فيه dept_id = 20 بس مفيش اسم القسم. عشان نجيب اسم القسم لازم نروح لجدول الأقسام.

المشكلة بدون JOIN
-- ده بيعرض dept_id = 20 بس مش بيقولنا اسم القسم! SELECT first_name, salary, dept_id FROM employees; -- عايزين بدل dept_id = 20 يكتب "التقنية" -- الحل: JOIN!
🎯 الـ JOIN بيربط جدولين (أو أكتر) عن طريق عمود مشترك بينهم. في حالتنا: employees.dept_id = departments.dept_id

أنواع الـ JOINs

INNER JOIN
بس اللي ليه match في الجدولين
LEFT JOIN
كل الشمال + المطابق من اليمين
RIGHT JOIN
كل اليمين + المطابق من الشمال
FULL OUTER JOIN
كل حاجة من الجدولين

Table Aliases - الاختصارات

لما بنربط جداول، لازم نحدد كل عمود جاي من أنهي جدول. عشان منكتبش اسم الجدول كامل كل مرة، بنستخدم Alias (اختصار).

بدون Alias vs مع Alias
-- ❌ طويل ومتعب: SELECT employees.first_name, departments.dept_name FROM employees INNER JOIN departments ON employees.dept_id = departments.dept_id; -- ✅ مع Alias (e و d): SELECT e.first_name, d.dept_name FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id;
💡 العادة: e للـ employees، d للـ departments، p للـ projects. اختار أي حرف يفكرك بالجدول.

INNER JOIN - التقاطع

بيرجع بس الصفوف اللي ليها match في الجدولين. لو صف مش ليه match - مش هيظهر خالص.

📐 الصيغة
SELECT columns FROM table1 alias1 INNER JOIN table2 alias2 ON alias1.column = alias2.column;
كل موظف مع اسم قسمه
SELECT e.emp_id, e.first_name, e.last_name, e.salary, d.dept_name, d.location FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id ORDER BY e.emp_id;

الموظفين في القاهرة بس اللي مرتبهم فوق 7000

INNER JOIN + GROUP BY - تقارير

تقرير: اسم كل قسم وعدد موظفينه ومتوسط مرتبه
SELECT d.dept_name, COUNT(e.emp_id) AS emp_count, SUM(e.salary) AS total_salary, ROUND(AVG(e.salary)) AS avg_salary, MAX(e.salary) AS max_salary FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id GROUP BY d.dept_name ORDER BY total_salary DESC;

LEFT JOIN - كل الشمال

بيرجع كل صفوف الجدول الأول (الشمال) + البيانات المطابقة من الجدول التاني. لو مفيش match بيحط NULL.

كل الموظفين - حتى اللي مش في قسم
SELECT e.emp_id, e.first_name, e.salary, d.dept_name, d.location FROM employees e LEFT JOIN departments d ON e.dept_id = d.dept_id ORDER BY e.emp_id;

إيجاد البيانات المفقودة

من أهم استخدامات LEFT JOIN: إيجاد الصفوف اللي مالهاش match. بنعمل LEFT JOIN وبعدين WHERE ... IS NULL.

الموظفين اللي مش في أي قسم
SELECT e.first_name, e.last_name, e.salary FROM employees e LEFT JOIN departments d ON e.dept_id = d.dept_id WHERE d.dept_id IS NULL; -- النتيجة: منى عبدالله - 11000
الأقسام اللي مفيهاش موظفين
-- لاحظ: هنا departments على الشمال! SELECT d.dept_name, d.location FROM departments d LEFT JOIN employees e ON d.dept_id = e.dept_id WHERE e.emp_id IS NULL; -- النتيجة: التسويق - المنصورة
🎯 ده Pattern مهم جداً في الشغل: LEFT JOIN + WHERE IS NULL = إيجاد المفقودين!
➡️

RIGHT JOIN - كل اليمين

عكس LEFT JOIN بالظبط. بيرجع كل صفوف الجدول التاني (اليمين) + المطابق من الأول.

FULL OUTER JOIN - كل حاجة

بيرجع كل الصفوف من الجدولين. اللي ليه match بيظهر كامل، واللي مالوش match بيظهر مع NULL.

كأنك عملت LEFT JOIN + RIGHT JOIN في نفس الوقت!

كل الموظفين وكل الأقسام
SELECT e.first_name, e.salary, d.dept_name, d.location FROM employees e FULL OUTER JOIN departments d ON e.dept_id = d.dept_id ORDER BY e.emp_id;

SELF JOIN - ربط الجدول بنفسه

أحياناً بنحتاج نربط الجدول بنفسه! مثلاً: كل موظف عنده manager_id - ده emp_id للمدير بتاعه في نفس الجدول.

كل موظف مع اسم مديره
SELECT e.first_name AS employee_name, e.salary AS employee_salary, m.first_name AS manager_name FROM employees e LEFT JOIN employees m -- نفس الجدول! ON e.manager_id = m.emp_id ORDER BY e.emp_id;

نصائح مهمة

🎯 قواعد ذهبية للـ JOINs
  • دايماً استخدم Aliases (e, d, p) عشان الكود يبقى نضيف ومقروء
  • حدد الأعمدة بالاسم - متستخدمش SELECT * مع JOINs
  • لو فيه عمود بنفس الاسم في الجدولين لازم تحدد الـ alias (مثلاً e.dept_id مش dept_id لوحدها)
  • INNER JOIN هو الافتراضي - لو كتبت JOIN بدون inner هيعتبرها INNER
  • LEFT JOIN هو الأكثر استخداماً في الشغل الحقيقي
  • الترتيب مهم: في LEFT JOIN، الجدول الشمال هو اللي كل صفوفه هتظهر
🏆 نصيحة أخيرة: لو مش متأكد تستخدم أنهي JOIN، ابدأ بـ INNER JOIN وشوف النتيجة. لو ناقص حاجة، جرب LEFT JOIN.

تمارين عملية

سهل 1: INNER JOIN أساسي◀

اعرض اسم كل موظف واسم قسمه ومكان القسم.

SELECT e.first_name, e.last_name, d.dept_name, d.location FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id;
سهل 2: INNER JOIN + WHERE◀

اعرض موظفين قسم التقنية بس مع مرتباتهم. رتب تنازلياً.

SELECT e.first_name, e.salary, d.dept_name FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id WHERE d.dept_name = 'التقنية' ORDER BY e.salary DESC;
سهل 3: LEFT JOIN◀

اعرض كل الموظفين مع اسم قسمهم (حتى اللي مش في قسم).

SELECT e.first_name, e.salary, d.dept_name FROM employees e LEFT JOIN departments d ON e.dept_id = d.dept_id;
متوسط 4: إيجاد المفقودين◀

اعرض الأقسام اللي مفيهاش أي موظف.

SELECT d.dept_name, d.location FROM departments d LEFT JOIN employees e ON d.dept_id = e.dept_id WHERE e.emp_id IS NULL;
متوسط 5: JOIN + GROUP BY◀

اعرض اسم كل قسم وعدد الموظفين فيه ومتوسط مرتبه (حتى الأقسام الفاضية).

SELECT d.dept_name, COUNT(e.emp_id) AS emp_count, ROUND(AVG(e.salary)) AS avg_salary FROM departments d LEFT JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_name ORDER BY emp_count DESC;
متوسط 6: JOIN + HAVING◀

اعرض الأقسام اللي فيها أكتر من 2 موظف ومتوسط مرتبهم.

SELECT d.dept_name, COUNT(e.emp_id) AS emp_count, ROUND(AVG(e.salary)) AS avg_salary FROM departments d INNER JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_name HAVING COUNT(e.emp_id) > 2;
متوسط 7: SELF JOIN◀

اعرض اسم كل موظف واسم مديره. اعرض "المدير العام" بدل NULL.

SELECT e.first_name AS employee, NVL(m.first_name, 'المدير العام') AS manager FROM employees e LEFT JOIN employees m ON e.manager_id = m.emp_id;
صعب 8: 3 جداول◀

اعرض اسم كل مشروع واسم القسم وعدد الموظفين في القسم ده.

SELECT p.project_name, d.dept_name, COUNT(e.emp_id) AS emp_count FROM projects p INNER JOIN departments d ON p.dept_id = d.dept_id LEFT JOIN employees e ON d.dept_id = e.dept_id GROUP BY p.project_name, d.dept_name ORDER BY emp_count DESC;
صعب 9: FULL OUTER◀

اعرض كل الموظفين وكل الأقسام. استخدم NVL عشان تعرض "بدون قسم" و"قسم فاضي" بدل NULL.

SELECT NVL(e.first_name, '--- قسم فاضي ---') AS employee, NVL(d.dept_name, 'بدون قسم') AS department FROM employees e FULL OUTER JOIN departments d ON e.dept_id = d.dept_id;
صعب 10: الموظفين أغنى من مديرهم◀

اعرض الموظفين اللي مرتبهم أعلى من مرتب مديرهم. اعرض اسم الموظف ومرتبه واسم المدير ومرتبه.

SELECT e.first_name AS employee, e.salary AS emp_salary, m.first_name AS manager, m.salary AS mgr_salary, e.salary - m.salary AS difference FROM employees e INNER JOIN employees m ON e.manager_id = m.emp_id WHERE e.salary > m.salary ORDER BY difference DESC;
تحدي 11: تقرير شامل◀

اعرض تقرير فيه: اسم القسم، المكان، عدد الموظفين، إجمالي المرتبات، عدد المشاريع. للأقسام اللي فيها أكتر من موظف.

SELECT d.dept_name, d.location, COUNT(DISTINCT e.emp_id) AS emp_count, SUM(DISTINCT e.salary) AS total_salary, COUNT(DISTINCT p.project_id) AS project_count FROM departments d INNER JOIN employees e ON d.dept_id = e.dept_id LEFT JOIN projects p ON d.dept_id = p.dept_id GROUP BY d.dept_name, d.location HAVING COUNT(DISTINCT e.emp_id) > 1 ORDER BY total_salary DESC;
تحدي 12: التحدي النهائي◀

اعرض تقرير لكل موظف فيه: اسمه، مرتبه، اسم قسمه، مكان القسم، اسم مديره، مرتب مديره. حتى لو الموظف مش في قسم أو مالوش مدير. رتب حسب المرتب تنازلياً.

SELECT e.first_name AS employee, e.salary AS salary, NVL(d.dept_name, 'بدون قسم') AS department, NVL(d.location, '-') AS location, NVL(m.first_name, 'المدير العام') AS manager, m.salary AS manager_salary FROM employees e LEFT JOIN departments d ON e.dept_id = d.dept_id LEFT JOIN employees m ON e.manager_id = m.emp_id ORDER BY e.salary DESC;
