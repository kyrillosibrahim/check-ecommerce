إيه هي الدوال التجميعية؟

الدوال التجميعية (Aggregate Functions) بتاخد مجموعة صفوف وترجع نتيجة واحدة.

فكر فيها زي دوال Excel: لما بتكتب =SUM(A1:A100) بتاخد 100 خلية وترجع رقم واحد.

🔢

COUNT - عدّ الصفوف بالتفصيل

COUNT ليها 3 أشكال مختلفة:
  • COUNT(*) → بيعد كل الصفوف بما فيها اللي فيها NULL
  • COUNT(column) → بيعد الصفوف اللي العمود فيها مش NULL
  • COUNT(DISTINCT column) → بيعد القيم المختلفة بس
COUNT(*) - عد كل الصفوف
-- كام موظف عندنا؟ SELECT COUNT(*) AS total_employees FROM employees; -- النتيجة: 10 (عد كل الصفوف حتى اللي فيها NULL)
COUNT(column) - تجاهل NULL
-- كام موظف عنده مرتب؟ (ريم مرتبها NULL) SELECT COUNT(salary) AS with_salary FROM employees; -- النتيجة: 9 (ريم مش محسوبة) -- كام موظف مرتبط بقسم؟ SELECT COUNT(dept_id) AS with_dept FROM employees; -- النتيجة: 9 (منى مش محسوبة لأن dept_id = NULL)
COUNT(DISTINCT) - القيم المختلفة بس
-- كام قسم مختلف عندنا موظفين فيه؟ SELECT COUNT(DISTINCT dept_id) AS dept_count FROM employees; -- النتيجة: 3 (أقسام 10, 20, 30 - بدون NULL)
COUNT مع WHERE
-- كام موظف مرتبه فوق 10000؟ SELECT COUNT(*) AS high_salary_count FROM employees WHERE salary > 10000; -- النتيجة: 3 (سارة 12000 + فاطمة 15000 + منى 11000)
💰

SUM - مجموع القيم

SUM بيجمع كل الأرقام في عمود معين. بيشتغل مع الأرقام بس (مش نصوص أو تواريخ).

أمثلة SUM
-- إجمالي كل المرتبات SELECT SUM(salary) AS total_salaries FROM employees; -- النتيجة: 79,000 (ريم متحسبتش لأن مرتبها NULL) -- إجمالي مرتبات قسم التقنية (20) SELECT SUM(salary) AS tech_total FROM employees WHERE dept_id = 20; -- النتيجة: 36,500 (12000 + 15000 + 9500) -- إجمالي مرتبات الموظفين الجداد (2024) SELECT SUM(salary) AS new_hires_total FROM employees WHERE hire_date >= DATE '2024-01-01'; -- النتيجة: 17,000 (عمر 7000 + يوسف 5500 + خالد 4500)
⚠️ SUM بيتجاهل NULL تماماً. يعني ريم (مرتبها NULL) مش بتأثر على المجموع نهائياً.
📈

AVG - المتوسط الحسابي

AVG = SUM ÷ COUNT (بس بيتجاهل NULL في الاتنين!)

أمثلة AVG
-- متوسط المرتبات SELECT AVG(salary) AS avg_salary FROM employees; -- النتيجة: 8777.777... (79000 ÷ 9 مش ÷ 10!) -- ROUND عشان تنظف الرقم SELECT ROUND(AVG(salary), 2) AS avg_salary FROM employees; -- النتيجة: 8777.78 -- بدون خانات عشرية SELECT ROUND(AVG(salary)) AS avg_salary FROM employees; -- النتيجة: 8778
🚨 مهم جداً: AVG بيقسم على 9 مش 10! لأنه بيتجاهل ريم (NULL). لو عايز تحسب NULL كـ 0 استخدم NVL:
NVL - تحويل NULL لقيمة
-- AVG مع احتساب NULL كـ 0 SELECT ROUND(AVG(salary)) AS avg_without_null, -- 8778 (÷9) ROUND(AVG(NVL(salary, 0))) AS avg_with_null -- 7900 (÷10) FROM employees; -- NVL(salary, 0) يعني: لو salary = NULL حطها 0
📊

MAX و MIN - أعلى وأقل قيمة

بيشتغلوا مع أي نوع بيانات: أرقام، نصوص (ترتيب أبجدي)، وتواريخ!

MAX و MIN مع أرقام
SELECT MAX(salary) AS highest, -- 15000 (فاطمة) MIN(salary) AS lowest, -- 4500 (خالد) MAX(salary) - MIN(salary) AS salary_range -- 10500 FROM employees;
MAX و MIN مع تواريخ
SELECT MAX(hire_date) AS newest_hire, -- 2024-08-01 (ريم) MIN(hire_date) AS oldest_hire -- 2021-03-10 (فاطمة) FROM employees;
MAX و MIN مع نصوص (أبجدي)
SELECT MAX(first_name) AS last_alpha, MIN(first_name) AS first_alpha FROM employees;

GROUP BY - تجميع البيانات

من غير GROUP BY، الدوال التجميعية بتشتغل على كل الجدول. بس لو عايز تعرف المتوسط لكل قسم لوحده؟ هنا بنستخدم GROUP BY!

زي ما في Excel بتعمل Pivot Table - بتقسم البيانات لمجموعات وتحسب لكل مجموعة.

📐 الصيغة
SELECT grouping_column, AGG_FUNCTION(column) FROM table [WHERE condition] GROUP BY grouping_column [ORDER BY ...];
🚨 قاعدة مهمة جداً: أي عمود في SELECT لازم يكون إما (1) جوه دالة تجميعية أو (2) موجود في GROUP BY. غير كده هيطلع Error!
مثال الخطأ الشائع
-- ❌ ده غلط! first_name مش في GROUP BY ومش في دالة SELECT dept_id, first_name, AVG(salary) FROM employees GROUP BY dept_id; -- ORA-00979: not a GROUP BY expression -- ✅ الصح: SELECT dept_id, AVG(salary) FROM employees GROUP BY dept_id;
📊

تجميع بأكتر من عمود

ممكن تجمّع بأكتر من عمود. مثلاً: عدد الموظفين في كل قسم وسنة تعيين.

GROUP BY عمودين
-- عدد الموظفين حسب القسم وسنة التعيين SELECT dept_id, EXTRACT(YEAR FROM hire_date) AS hire_year, COUNT(*) AS emp_count FROM employees WHERE dept_id IS NOT NULL GROUP BY dept_id, EXTRACT(YEAR FROM hire_date) ORDER BY dept_id, hire_year;
💡 EXTRACT(YEAR FROM date) بيطلع السنة من التاريخ. ممكن كمان MONTH أو DAY.

HAVING - فلترة المجموعات

WHERE بيفلتر الصفوف قبل التجميع. بس لو عايز تفلتر نتيجة التجميع؟ مثلاً "الأقسام اللي فيها أكتر من 2 موظف" - مش هتقدر بـ WHERE!

HAVING بيفلتر المجموعات بعد التجميع.

الأقسام اللي فيها أكتر من 2 موظف
SELECT dept_id, COUNT(*) AS emp_count FROM employees GROUP BY dept_id HAVING COUNT(*) > 2;

HAVING مع دوال مختلفة
-- الأقسام اللي متوسط مرتبها فوق 8000 SELECT dept_id, ROUND(AVG(salary)) AS avg_salary FROM employees GROUP BY dept_id HAVING AVG(salary) > 8000; -- الأقسام اللي إجمالي مرتباتها فوق 20000 SELECT dept_id, SUM(salary) AS total_salary FROM employees GROUP BY dept_id HAVING SUM(salary) > 20000;
WHERE + GROUP BY + HAVING معاً
-- الأقسام اللي فيها أكتر من موظف واحد -- بس نحسب بس الموظفين اللي اتعينوا بعد 2022 -- واللي متوسط مرتبهم فوق 6000 SELECT dept_id, COUNT(*) AS emp_count, ROUND(AVG(salary)) AS avg_salary FROM employees WHERE hire_date >= DATE '2022-01-01' -- 1. فلتر الصفوف GROUP BY dept_id -- 2. جمّع HAVING COUNT(*) > 1 -- 3. فلتر المجموعات AND AVG(salary) > 6000 ORDER BY avg_salary DESC; -- 4. رتّب
⚔️

WHERE vs HAVING - إيه الفرق؟

📋 WHERE

بيفلتر الصفوف

بيشتغل قبل GROUP BY

مينفعش يستخدم دوال تجميعية

مثال: WHERE salary > 5000

🎯 HAVING

بيفلتر المجموعات

بيشتغل بعد GROUP BY

لازم يستخدم دوال تجميعية

مثال: HAVING AVG(salary) > 5000

مثال يوضح الفرق
-- ❌ غلط: WHERE مش بيشتغل مع دوال تجميعية SELECT dept_id, AVG(salary) FROM employees WHERE AVG(salary) > 8000 -- ERROR! GROUP BY dept_id; -- ✅ صح: HAVING بيشتغل مع دوال تجميعية SELECT dept_id, ROUND(AVG(salary)) AS avg_sal FROM employees GROUP BY dept_id HAVING AVG(salary) > 8000; -- OK!

ترتيب تنفيذ الـ Query

Oracle مش بينفذ الـ SQL بالترتيب اللي انت كاتبه! الترتيب الحقيقي مختلف:

1. FROM
←
2. WHERE
←
3. GROUP BY
←
4. HAVING
←
5. SELECT

6. ORDER BY
📖 يعني إيه ده؟
  • FROM: حدد الجدول الأول
  • WHERE: فلتر الصفوف (عشان كده مينفعش تستخدم Alias هنا)
  • GROUP BY: قسّم الصفوف لمجموعات
  • HAVING: فلتر المجموعات
  • SELECT: اختار الأعمدة والدوال
  • ORDER BY: رتّب النتيجة النهائية

تمارين عملية

سهل 1: COUNT الأساسي ◀

اعرض عدد الموظفين الكلي وعدد الموظفين اللي عندهم مرتب.

SELECT COUNT(*) AS total, COUNT(salary) AS with_salary FROM employees;
سهل 2: SUM و AVG ◀

اعرض إجمالي ومتوسط المرتبات لقسم المبيعات (dept_id = 10).

SELECT SUM(salary) AS total_salary, ROUND(AVG(salary)) AS avg_salary FROM employees WHERE dept_id = 10;
سهل 3: MAX و MIN ◀

اعرض أعلى مرتب، أقل مرتب، والفرق بينهم.

SELECT MAX(salary) AS highest, MIN(salary) AS lowest, MAX(salary) - MIN(salary) AS diff FROM employees;
متوسط 4: GROUP BY بسيط ◀

اعرض لكل قسم: عدد الموظفين ومتوسط المرتب. رتب حسب المتوسط تنازلياً.

SELECT dept_id, COUNT(*) AS emp_count, ROUND(AVG(salary)) AS avg_salary FROM employees GROUP BY dept_id ORDER BY avg_salary DESC;
متوسط 5: GROUP BY مع WHERE ◀

اعرض إجمالي المرتبات لكل قسم، بس احسب الموظفين اللي مرتبهم فوق 5000 بس.

SELECT dept_id, SUM(salary) AS total_salary FROM employees WHERE salary > 5000 GROUP BY dept_id ORDER BY total_salary DESC;
متوسط 6: HAVING ◀

اعرض الأقسام اللي فيها أكتر من موظفين اتنين.

SELECT dept_id, COUNT(*) AS emp_count FROM employees GROUP BY dept_id HAVING COUNT(*) > 2;
صعب 7: WHERE + HAVING ◀

اعرض الأقسام اللي متوسط مرتبها فوق 7000، بس احسب بس الموظفين اللي اتعينوا في 2023 أو بعدها.

SELECT dept_id, COUNT(*) AS emp_count, ROUND(AVG(salary)) AS avg_salary FROM employees WHERE hire_date >= DATE '2023-01-01' GROUP BY dept_id HAVING AVG(salary) > 7000 ORDER BY avg_salary DESC;
صعب 8: NVL مع AVG ◀

اعرض متوسط المرتبات مرتين: مرة بتجاهل NULL ومرة باحتساب NULL كـ 0.

SELECT ROUND(AVG(salary)) AS avg_ignore_null, ROUND(AVG(NVL(salary, 0))) AS avg_null_as_zero FROM employees;
صعب 9: تقرير شامل ◀

اعرض لكل قسم: عدد الموظفين، إجمالي المرتبات، المتوسط، أعلى مرتب، أقل مرتب. بس الأقسام اللي إجمالي مرتباتها فوق 10000. رتب تنازلياً.

SELECT dept_id, COUNT(*) AS emp_count, SUM(salary) AS total_sal, ROUND(AVG(salary)) AS avg_sal, MAX(salary) AS max_sal, MIN(salary) AS min_sal FROM employees GROUP BY dept_id HAVING SUM(salary) > 10000 ORDER BY total_sal DESC;
تحدي 10: التحدي النهائي ◀

اكتب Query واحد يعرض: عدد كل الموظفين، عدد اللي عندهم مرتب، عدد الأقسام المختلفة، إجمالي المرتبات، المتوسط (مع NULL كـ 0)، أعلى مرتب، أقل مرتب، الفرق بين أعلى وأقل.

SELECT COUNT(*) AS total_emp, COUNT(salary) AS with_salary, COUNT(DISTINCT dept_id) AS dept_count, SUM(salary) AS total_sal, ROUND(AVG(NVL(salary,0))) AS avg_sal, MAX(salary) AS max_sal, MIN(salary) AS min_sal, MAX(salary) - MIN(salary) AS sal_range FROM employees;
