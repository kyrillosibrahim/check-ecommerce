UPDATE - تعديل البيانات

📐 الصيغة الأساسية
UPDATE table_name SET column1 = value1, column2 = value2, ... WHERE condition;
🚨 تحذير مهم جداً: لو نسيت الـ WHERE هيتعدل كل الصفوف في الجدول! دايماً اكتب WHERE الأول.
تعديل عمود واحد
تعديل مرتب أحمد
-- زيادة مرتب أحمد لـ 9000 UPDATE employees SET salary = 9000 WHERE emp_id = 101;
📊 النتيجة بعد التعديل
emp_idfirst_namesalary قبلsalary بعد
101أحمد8,0009,000 ✅
تعديل أكثر من عمود
تعديل المرتب والإيميل معاً
UPDATE employees SET salary = 7500, email = 'omar.new@mail.com' WHERE emp_id = 105;
تعديل بناءً على شرط
زيادة 10% لكل موظفين قسم المبيعات
-- زيادة 10% لكل موظفين قسم 10 (المبيعات) UPDATE employees SET salary = salary * 1.10 WHERE dept_id = 10;
تعديل بإستخدام NULL
تعيين منى لقسم التسويق
-- منى كانت مش في أي قسم، نضيفها للتسويق UPDATE employees SET dept_id = 40 WHERE emp_id = 108;
💡 نصيحة: دايماً اعمل SELECT الأول عشان تشوف الصفوف اللي هتتأثر، وبعدين حوّلها لـ UPDATE.
🗑️

DELETE - حذف البيانات

📐 الصيغة الأساسية
DELETE FROM table_name WHERE condition;
🚨 تحذير: لو نسيت WHERE هيتحذف كل البيانات من الجدول! ده أخطر أمر في SQL.
حذف صف واحد
حذف موظف معين
-- حذف يوسف (107) DELETE FROM employees WHERE emp_id = 107;
حذف بناءً على شرط
حذف الموظفين اللي مرتبهم أقل من 6000
-- شوف الأول مين هيتحذف SELECT * FROM employees WHERE salary < 6000; -- لو النتيجة صح، احذف DELETE FROM employees WHERE salary < 6000;
حذف كل البيانات
⚠️ حذف كل البيانات - خطير!
-- ⚠️ ده هيحذف كل الموظفين! DELETE FROM employees; -- ممكن ترجع البيانات لو معملتش COMMIT ROLLBACK;
💡 ROLLBACK بيرجع البيانات لو لسه معملتش COMMIT. دايماً استخدم COMMIT بعد ما تتأكد إن كل حاجة صح.

ALTER TABLE - تعديل هيكل الجدول

ALTER TABLE بيعدل على هيكل الجدول نفسه (أعمدة، أنواع بيانات، constraints) - مش البيانات اللي جواه.

إضافة عمود جديد
ADD - إضافة عمود
-- إضافة عمود رقم التليفون ALTER TABLE employees ADD phone VARCHAR2(15); -- إضافة عمود بقيمة افتراضية ALTER TABLE employees ADD status VARCHAR2(10) DEFAULT 'Active';
تعديل عمود موجود
MODIFY - تعديل نوع أو حجم العمود
-- تكبير حجم عمود الإيميل ALTER TABLE employees MODIFY email VARCHAR2(100); -- جعل عمود إجباري ALTER TABLE employees MODIFY phone NOT NULL;
إعادة تسمية عمود
RENAME COLUMN
ALTER TABLE employees RENAME COLUMN phone TO mobile;
حذف عمود
DROP COLUMN - حذف عمود
ALTER TABLE employees DROP COLUMN mobile;
⚠️ ALTER TABLE هو DDL (لغة تعريف البيانات) - مش DML. يعني بيعمل COMMIT تلقائي ومش بتقدر تعمل ROLLBACK!
💥

DROP TABLE - حذف الجدول بالكامل

📐 الصيغة
DROP TABLE table_name; -- لو فيه Foreign Key بتشير للجدول ده DROP TABLE table_name CASCADE CONSTRAINTS;
⚡ الفرق بين DELETE و DROP
  • DELETE: بيحذف البيانات (الصفوف) بس - الجدول يفضل موجود
  • DROP: بيحذف الجدول بالكامل - الهيكل والبيانات والـ indexes وكل حاجة
أمثلة
-- حذف جدول (لو مفيش FK بيشير له) DROP TABLE temp_table; -- حذف جدول حتى لو فيه FK DROP TABLE departments CASCADE CONSTRAINTS; -- حذف مع وضعه في سلة المحذوفات (Oracle) DROP TABLE old_data; -- ممكن ترجعه بـ: FLASHBACK TABLE old_data TO BEFORE DROP;
💀 DROP TABLE عملية لا رجعة فيها (إلا بالـ Flashback في Oracle). فكر مرتين قبل ما تستخدمها!
⚔️

TRUNCATE - حذف كل البيانات بسرعة
-- حذف كل البيانات من الجدول (مش بيتحذف الجدول نفسه) TRUNCATE TABLE temp_data; -- الفرق: DELETE بيمسح صف صف (بطيء + log) -- TRUNCATE بيمسح كل حاجة مرة واحدة (سريع + بدون log)

الدوال التجميعية (Aggregate Functions)

الدوال التجميعية بتاخد مجموعة قيم وترجع قيمة واحدة. زي ما Excel فيه SUM و AVERAGE، Oracle عنده نفس الحاجة!

💡 كل الدوال التجميعية بتتجاهل الـ NULL تلقائياً (ما عدا COUNT(*)).
🔢

COUNT - عدد الصفوف

أنواع COUNT المختلفة
-- عدد كل الموظفين SELECT COUNT(*) AS total_employees FROM employees; -- النتيجة: 8 -- عدد الموظفين اللي ليهم قسم SELECT COUNT(dept_id) AS with_dept FROM employees; -- النتيجة: 7 (منى مش محسوبة لأن dept_id = NULL) -- عدد الأقسام المختلفة SELECT COUNT(DISTINCT dept_id) AS dept_count FROM employees; -- النتيجة: 3 (أقسام 10, 20, 30)
الفرق بين COUNT(*) و COUNT(column)
  • COUNT(*) → بيعد كل الصفوف حتى اللي فيها NULL
  • COUNT(column) → بيعد الصفوف اللي العمود مش NULL فيها
  • COUNT(DISTINCT column) → بيعد القيم المختلفة بس
💰

SUM و AVG

SUM - مجموع المرتبات
-- إجمالي المرتبات SELECT SUM(salary) AS total_salaries FROM employees; -- النتيجة: 74,500 -- إجمالي مرتبات قسم التقنية SELECT SUM(salary) AS tech_salaries FROM employees WHERE dept_id = 20; -- النتيجة: 36,500
AVG - متوسط المرتبات
-- متوسط المرتبات SELECT ROUND(AVG(salary), 2) AS avg_salary FROM employees; -- النتيجة: 9,312.50 -- استخدام SUM و AVG و COUNT معاً SELECT COUNT(*) AS employees_count, SUM(salary) AS total_salary, ROUND(AVG(salary)) AS avg_salary FROM employees;
💡 ROUND(value, 2) بيقرّب الرقم لأقرب خانتين عشريتين. بنستخدمها كتير مع AVG.
📈

MAX و MIN

أعلى وأقل قيمة
-- أعلى وأقل مرتب SELECT MAX(salary) AS highest_salary, MIN(salary) AS lowest_salary, MAX(salary) - MIN(salary) AS salary_range FROM employees; -- النتيجة: 15000, 5500, 9500 -- أحدث وأقدم تاريخ تعيين SELECT MAX(hire_date) AS newest_hire, MIN(hire_date) AS oldest_hire FROM employees; -- MAX و MIN بيشتغلوا مع النصوص كمان (ترتيب أبجدي) SELECT MAX(first_name) AS last_alphabetically, MIN(first_name) AS first_alphabetically FROM employees;

GROUP BY - تجميع البيانات

GROUP BY بيقسم البيانات لمجموعات وبيطبق الدوال التجميعية على كل مجموعة لوحدها.

زي ما في Excel بتعمل Pivot Table - بتجمع البيانات حسب فئة معينة.

📐 الصيغة
SELECT column, AGG_FUNCTION(column) FROM table GROUP BY column;
عدد الموظفين في كل قسم
SELECT dept_id, COUNT(*) AS emp_count FROM employees GROUP BY dept_id;
📊 النتيجة

NULL1
إحصائيات كاملة لكل قسم
SELECT dept_id, COUNT(*) AS emp_count, SUM(salary) AS total_salary, ROUND(AVG(salary)) AS avg_salary, MAX(salary) AS max_salary, MIN(salary) AS min_salary FROM employees GROUP BY dept_id ORDER BY dept_id;
⚠️ قاعدة مهمة: أي عمود في SELECT لازم يكون إما جوه دالة تجميعية أو موجود في GROUP BY. مينفعش تكتب first_name مع GROUP BY dept_id!
🎯

HAVING - فلترة المجموعات

❓ إيه الفرق بين WHERE و HAVING؟
  • WHERE: بيفلتر الصفوف قبل التجميع
  • HAVING: بيفلتر المجموعات بعد التجميع
الأقسام اللي فيها أكتر من موظف واحد
SELECT dept_id, COUNT(*) AS emp_count FROM employees GROUP BY dept_id HAVING COUNT(*) > 1;
WHERE + GROUP BY + HAVING معاً
-- الأقسام اللي متوسط مرتبها فوق 8000 -- بس للموظفين اللي اتعينوا بعد 2022 SELECT dept_id, COUNT(*) AS emp_count, ROUND(AVG(salary)) AS avg_salary FROM employees WHERE hire_date >= DATE '2022-01-01' -- فلتر الصفوف الأول GROUP BY dept_id -- بعدين جمّع HAVING AVG(salary) > 8000 -- بعدين فلتر المجموعات ORDER BY avg_salary DESC; -- وأخيراً رتّب
📋 ترتيب التنفيذ: FROM → WHERE → GROUP BY → HAVING → SELECT → ORDER BY
🔗

مقدمة الـ JOINs - ربط الجداول

الـ JOIN هو أهم موضوع في SQL! بيسمحلك تربط بيانات من جدولين أو أكتر معاً باستخدام عمود مشترك بينهم.

مثلاً: جدول الموظفين فيه dept_id بس مفيش اسم القسم. عشان نجيب اسم القسم لازم نعمل JOIN مع جدول الأقسام.

🎯 تخيل عندك جدول تليفونات وجدول عناوين. الـ JOIN بيربطهم عشان تشوف كل شخص ومعاه رقمه وعنوانه في نتيجة واحدة.
🗂️ أنواع الـ JOINs
  • INNER JOIN: بس الصفوف اللي ليها match في الجدولين
  • LEFT JOIN: كل صفوف الجدول الشمال + اللي ليها match من اليمين
  • RIGHT JOIN: كل صفوف الجدول اليمين + اللي ليها match من الشمال
  • FULL OUTER JOIN: كل الصفوف من الجدولين
🎯

INNER JOIN - التقاطع

بيرجع بس الصفوف اللي ليها match في الجدولين. لو موظف مش في أي قسم (NULL) مش هيظهر.

📐 الصيغة
SELECT columns FROM table1 INNER JOIN table2 ON table1.column = table2.column;
الموظفين مع أسماء أقسامهم
SELECT e.first_name, e.last_name, e.salary, d.dept_name, d.location FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id;

LEFT JOIN - كل اللي على الشمال

بيرجع كل الصفوف من الجدول الأول (الشمال) + البيانات المطابقة من الجدول التاني. لو مفيش match، بيحط NULL.

ربط أكتر من جدولين

ممكن تربط 3 جداول أو أكتر في query واحد. كل JOIN بيضيف جدول جديد.

مثال: لو عندنا جدول مشاريع
-- إنشاء جدول المشاريع CREATE TABLE projects ( project_id NUMBER(4) PRIMARY KEY, project_name VARCHAR2(50), dept_id NUMBER(4) REFERENCES departments(dept_id) ); INSERT INTO projects VALUES (1, 'موقع التجارة الإلكترونية', 20); INSERT INTO projects VALUES (2, 'حملة التسويق الرقمي', 40); INSERT INTO projects VALUES (3, 'نظام المبيعات', 10); COMMIT; -- ربط 3 جداول: الموظفين + الأقسام + المشاريع SELECT e.first_name, d.dept_name, p.project_name FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id INNER JOIN projects p ON d.dept_id = p.dept_id;
JOIN مع GROUP BY - إحصائيات
-- عدد الموظفين ومتوسط المرتب لكل قسم SELECT d.dept_name, COUNT(e.emp_id) AS emp_count, ROUND(AVG(e.salary)) AS avg_salary FROM departments d LEFT JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_name ORDER BY emp_count DESC;
🎯 الـ JOIN مع GROUP BY + HAVING هي أقوى combination في SQL وهتستخدمها كتير في الشغل!

تمارين عملية

سهل تمرين 1: UPDATE ◀

زوّد مرتب كل الموظفين في قسم التقنية (20) بنسبة 15%.

UPDATE employees SET salary = salary * 1.15 WHERE dept_id = 20;
سهل تمرين 2: DELETE ◀

احذف الموظفين اللي مرتبهم أقل من 6000 واللي مش في أي قسم.

DELETE FROM employees WHERE salary < 6000 AND dept_id IS NULL;
متوسط تمرين 3: ALTER TABLE ◀

أضف عمود bonus (نوعه NUMBER) لجدول الموظفين بقيمة افتراضية 0.

ALTER TABLE employees ADD bonus NUMBER(10,2) DEFAULT 0;
متوسط تمرين 4: GROUP BY ◀

اعرض لكل قسم: عدد الموظفين، إجمالي المرتبات، أعلى مرتب. ورتب حسب إجمالي المرتبات تنازلياً.

SELECT dept_id, COUNT(*) AS emp_count, SUM(salary) AS total_salary, MAX(salary) AS max_salary FROM employees GROUP BY dept_id ORDER BY total_salary DESC;
متوسط تمرين 5: HAVING ◀

اعرض الأقسام اللي متوسط المرتب فيها أكبر من 8000.

SELECT dept_id, ROUND(AVG(salary)) AS avg_salary FROM employees GROUP BY dept_id HAVING AVG(salary) > 8000;
صعب تمرين 6: INNER JOIN ◀

اعرض اسم كل موظف واسم قسمه ومكان القسم، بس للموظفين اللي مرتبهم فوق 7000.

SELECT e.first_name, d.dept_name, d.location, e.salary FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id WHERE e.salary > 7000 ORDER BY e.salary DESC;
صعب تمرين 7: LEFT JOIN + GROUP BY ◀

اعرض اسم كل قسم وعدد الموظفين فيه (حتى لو القسم فاضي). رتب حسب عدد الموظفين تنازلياً.

SELECT d.dept_name, COUNT(e.emp_id) AS emp_count FROM departments d LEFT JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_name ORDER BY emp_count DESC;
صعب تمرين 8: التحدي النهائي ◀

اعرض تقرير فيه: اسم القسم، عدد الموظفين، متوسط المرتب، أعلى مرتب. بس للأقسام اللي فيها أكتر من موظف واحد ومتوسط المرتب فيها فوق 7000. رتب حسب متوسط المرتب تنازلياً.

SELECT d.dept_name, COUNT(e.emp_id) AS emp_count, ROUND(AVG(e.salary)) AS avg_salary, MAX(e.salary) AS max_salary FROM departments d INNER JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_name HAVING COUNT(e.emp_id) > 1 AND AVG(e.salary) > 7000 ORDER BY avg_salary DESC;
