مقدمة PL/SQL

هتتعلم لغة البرمجة الخاصة بـ Oracle! من المتغيرات والـ BEGIN...END لحد ما تكتب أول برنامج بتاعك.

المقدمة

إيه هو PL/SQL؟

PL/SQL = Procedural Language / SQL

هي لغة برمجة خاصة بـ Oracle بتضيف قدرات البرمجة (زي المتغيرات والـ loops والـ conditions) على الـ SQL العادي. يعني بدل ما تكتب SQL statement واحد، تقدر تكتب برنامج كامل يشتغل جوه الداتابيز!

🧠

تشبيه: SQL = الأوامر، PL/SQL = البرنامج!

SQL زي إنك بتدي أوامر فردية: "هات البيانات"، "حدّث المرتب". لكن PL/SQL زي إنك بتكتب سيناريو كامل: "لو المرتب أقل من 5000، زوّده 10%، ولو أكتر اطبع رسالة".

مقارنة

SQL vs PL/SQL

📝 SQL

  • لغة استعلام (Query Language)
  • أمر واحد في كل مرة
  • مفيهاش متغيرات
  • مفيهاش شروط أو loops
  • بتتنفذ statement بـ statement
VS

⚡ PL/SQL

  • لغة برمجة كاملة
  • بلوك كامل من الأوامر
  • فيها متغيرات وثوابت
  • فيها IF/ELSE و LOOP
  • بتتنفذ كبلوك واحد على الـ server
✅ ميزة الأداء: PL/SQL بتبعت البلوك كله مرة واحدة للـ server بدل ما تبعت كل أمر لوحده. ده بيقلل الـ network traffic وبيحسن الأداء!

هيكل الـ PL/SQL Block

كل برنامج PL/SQL بيتكون من Block ليه 3 أجزاء:

اختياري DECLARE
→
إجباري ✅ BEGIN...END
→
اختياري EXCEPTION
PL/SQL - Block Structure 
DECLARE
    -- هنا بتعرّف المتغيرات (اختياري)
    variable_name  datatype;

BEGIN
    -- هنا الكود الرئيسي (إجباري!)
    -- SQL statements + PL/SQL logic

EXCEPTION
    -- هنا بتتعامل مع الأخطاء (اختياري)
    WHEN error_name THEN
        -- handle error

END;
/
⚠️ مهم جداً: لازم تحط ; بعد كل statement ولازم تحط / في سطر لوحده بعد END; عشان SQL*Plus ينفذ البلوك!

المتغيرات

المتغيرات Variables

المتغير = صندوق بتخزن فيه قيمة وتستخدمها بعدين.

طريقة التعريف

PL/SQL - Variable Syntax 
-- الصيغة:
variable_name  datatype  [:= initial_value];

-- أمثلة:
v_name     VARCHAR2(50);             -- نص بدون قيمة أولية
v_salary   NUMBER(10,2) := 5000;     -- رقم بقيمة أولية
v_is_active BOOLEAN := TRUE;       -- قيمة منطقية
v_hire_date DATE := SYSDATE;       -- تاريخ اليوم
💡 اسم المتغير: العادة إنك تبدأ بـ v_ عشان تفرّقه عن أسماء الأعمدة. مثلاً: v_salary بدل salary.

مثال عملي كامل

PL/SQL - Variables Example 
DECLARE
    v_name    VARCHAR2(50) := 'أحمد محمد';
    v_salary  NUMBER(10,2) := 7500.50;
    v_bonus   NUMBER(10,2);
    v_total   NUMBER(10,2);
BEGIN
    -- حساب البونص (10% من المرتب)
    v_bonus := v_salary * 0.10;
    v_total := v_salary + v_bonus;

    DBMS_OUTPUT.PUT_LINE('الموظف: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('البونص: ' || v_bonus);
    DBMS_OUTPUT.PUT_LINE('الإجمالي: ' || v_total);
END;
/

النتيجة:

Output
الموظف: أحمد محمد
المرتب: 7500.5
البونص: 750.05
الإجمالي: 8250.55
لاحظ: علامة := هي علامة التعيين (assignment) في PL/SQL. مش = زي Java أو Python!

ذكي

%TYPE و %ROWTYPE

بدل ما تحفظ نوع كل عمود، خلي PL/SQL يحدده تلقائي!

%TYPE - نفس نوع عمود معين

PL/SQL - %TYPE 
DECLARE
    -- المتغير هياخد نفس نوع عمود salary في جدول employees
    v_salary  employees.salary%TYPE;
    v_name    employees.first_name%TYPE;
    v_dept_id employees.department_id%TYPE;
BEGIN
    SELECT first_name, salary, department_id
    INTO v_name, v_salary, v_dept_id
    FROM employees
    WHERE employee_id = 100;

    DBMS_OUTPUT.PUT_LINE(v_name || ' - ' || v_salary);
END;
/

%ROWTYPE - صف كامل من الجدول

PL/SQL - %ROWTYPE 
DECLARE
    -- متغير واحد فيه كل أعمدة الجدول!
    v_emp  employees%ROWTYPE;
BEGIN
    SELECT *
    INTO v_emp
    FROM employees
    WHERE employee_id = 100;

    DBMS_OUTPUT.PUT_LINE('الاسم: ' || v_emp.first_name);
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_emp.salary);
    DBMS_OUTPUT.PUT_LINE('القسم: ' || v_emp.department_id);
END;
/
✅ ليه %TYPE و %ROWTYPE أفضل؟
  • لو نوع العمود اتغير في الجدول، الكود بتاعك مش هيحتاج تعديل!
  • مش محتاج تحفظ إن salary هو NUMBER(10,2) - Oracle بيحدده لوحده
  • %ROWTYPE بتوفر عليك تعريف 10 متغيرات لـ 10 أعمدة
ثوابت

الثوابت Constants

الثابت زي المتغير بس قيمته مش بتتغير بعد التعريف:

PL/SQL - Constants 
DECLARE
    -- ثوابت
    c_tax_rate    CONSTANT NUMBER(4,2) := 0.14;  -- ضريبة 14%
    c_bonus_rate  CONSTANT NUMBER(4,2) := 0.10;  -- بونص 10%
    c_company     CONSTANT VARCHAR2(50) := 'شركة النجمة';

    v_salary  NUMBER := 8000;
    v_net     NUMBER;
BEGIN
    v_net := v_salary - (v_salary * c_tax_rate) + (v_salary * c_bonus_rate);

    DBMS_OUTPUT.PUT_LINE(c_company);
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('الصافي: ' || v_net);

    -- ده هيفشل ❌
    -- c_tax_rate := 0.20;  -- ORA-06550: cannot assign to a CONSTANT
END;
/
نصيحة: استخدم c_ كبداية لأسماء الثوابت عشان تفرّقها عن المتغيرات. c_tax_rate = ثابت، v_salary = متغير.

BEGIN

    DBMS_OUTPUT.PUT_LINE('النتيجة = ' || (100 + 200));

END;
/

سطر تاني
أحمد محمد
النتيجة = 300

متغيرات الإدخال (Substitution Variables)

في SQL*Plus تقدر تاخد input من المستخدم بعلامة &:

PL/SQL - Substitution Variable 
-- هيسألك تدخل الـ employee_id
DECLARE
    v_id     NUMBER := &enter_employee_id;
    v_name   employees.first_name%TYPE;
    v_salary employees.salary%TYPE;
BEGIN
    SELECT first_name, salary
    INTO v_name, v_salary
    FROM employees
    WHERE employee_id = v_id;

    DBMS_OUTPUT.PUT_LINE('الموظف: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
END;
/
⚠️ ملاحظة: الـ &variable بتشتغل في SQL*Plus و SQL Developer. في Oracle Live SQL ممكن تكون مش متاحة، فاستخدم قيمة ثابتة بدلها.
مهم جداً

SELECT INTO - جلب بيانات من الجدول

عشان تجيب بيانات من جدول وتحطها في متغيرات، استخدم SELECT ... INTO:

PL/SQL - SELECT INTO 
DECLARE
    v_name    employees.first_name%TYPE;
    v_salary  employees.salary%TYPE;
    v_dept    departments.department_name%TYPE;
BEGIN
    -- جلب بيانات صف واحد
    SELECT e.first_name, e.salary, d.department_name
    INTO v_name, v_salary, v_dept
    FROM employees e
    JOIN departments d ON e.department_id = d.department_id
    WHERE e.employee_id = 100;

    DBMS_OUTPUT.PUT_LINE('الاسم: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('القسم: ' || v_dept);
END;
/
🚨 أخطاء شائعة مع SELECT INTO:
الخطأالسببالحل
NO_DATA_FOUND الـ WHERE مرجعش أي صف تأكد إن البيانات موجودة
TOO_MANY_ROWS الـ Query رجّعت أكتر من صف استخدم WHERE أدق أو Cursor

حالات خاصة

التعامل مع NULL

في PL/SQL أي متغير مش اتحدد له قيمة بيكون NULL. وأي عملية مع NULL نتيجتها NULL!

PL/SQL - NULL Trap 
DECLARE
    v_bonus  NUMBER;  -- ده NULL! مفيش قيمة أولية
    v_salary NUMBER := 5000;
    v_total  NUMBER;
BEGIN
    v_total := v_salary + v_bonus;
    -- v_total = NULL! 😱 (5000 + NULL = NULL)

    DBMS_OUTPUT.PUT_LINE('الإجمالي: ' || v_total);
    -- Output: الإجمالي:    (فاضي!)

    -- الحل: استخدم NVL
    v_total := v_salary + NVL(v_bonus, 0);
    DBMS_OUTPUT.PUT_LINE('الإجمالي الصح: ' || v_total);
    -- Output: الإجمالي الصح: 5000 ✅
END;
/
🚨 قاعدة: أي عملية مع NULL = NULL!
  • NULL + 5000 = NULL
  • NULL || 'text' = 'text' (الاستثناء الوحيد: الدمج بـ ||)
  • NULL = NULL → ليست TRUE! (استخدم IS NULL)

تمرين 2: حاسبة المرتب

سهل

اكتب برنامج يحسب صافي المرتب. المرتب الأساسي 6000، البونص 15%، الضريبة 14%. اطبع المرتب الأساسي والبونص والضريبة والصافي.

الصافي = المرتب + البونص - الضريبة. البونص = مرتب × 0.15، الضريبة = مرتب × 0.14

PL/SQL
DECLARE
    v_salary NUMBER := 6000;
    v_bonus  NUMBER;
    v_tax    NUMBER;
    v_net    NUMBER;
BEGIN
    v_bonus := v_salary * 0.15;
    v_tax   := v_salary * 0.14;
    v_net   := v_salary + v_bonus - v_tax;

    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('البونص: ' || v_bonus);
    DBMS_OUTPUT.PUT_LINE('الضريبة: ' || v_tax);
    DBMS_OUTPUT.PUT_LINE('الصافي: ' || v_net);
END;
/

تمرين 6: حاسبة الضرائب المتقدمة

صعب

اكتب برنامج يحسب ضريبة الدخل. عرّف ثوابت للشرائح الضريبية:
- أول 5000 معفية (0%)
- من 5001 لـ 10000 ضريبة 10%
- أكتر من 10000 ضريبة 15%
المرتب = 12000. احسب الضريبة الإجمالية والصافي.

الضريبة = 0 (أول 5000) + 5000 × 10% (الشريحة التانية) + 2000 × 15% (الباقي)

PL/SQL
DECLARE
    c_bracket1 CONSTANT NUMBER := 5000;
    c_bracket2 CONSTANT NUMBER := 10000;
    c_rate1    CONSTANT NUMBER := 0.10;
    c_rate2    CONSTANT NUMBER := 0.15;

    v_salary NUMBER := 12000;
    v_tax    NUMBER := 0;
    v_net    NUMBER;
BEGIN
    -- أول 5000 معفية
    -- من 5001 لـ 10000 = 5000 × 10%
    v_tax := v_tax + (c_bracket2 - c_bracket1) * c_rate1;
    -- الباقي (12000 - 10000 = 2000) × 15%
    v_tax := v_tax + (v_salary - c_bracket2) * c_rate2;

    v_net := v_salary - v_tax;

    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('الضريبة: ' || v_tax);
    DBMS_OUTPUT.PUT_LINE('الصافي: ' || v_net);
END;
/
-- المرتب: 12000
-- الضريبة: 800  (500 + 300)
-- الصافي: 11200

تمرين 7: NVL مع SELECT INTO

صعب

اكتب برنامج يجيب اسم الموظف ومرتبه وعمولته (commission_pct) للموظف رقم 100. لو العمولة NULL اطبع "بدون عمولة" ولو فيها قيمة اطبع المبلغ (salary × commission_pct).

PL/SQL
DECLARE
    v_name employees.first_name%TYPE;
    v_salary employees.salary%TYPE;
    v_comm employees.commission_pct%TYPE;
BEGIN
    SELECT first_name, salary, commission_pct
    INTO v_name, v_salary, v_comm
    FROM employees
    WHERE employee_id = 100;

    DBMS_OUTPUT.PUT_LINE('الاسم: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);

    IF v_comm IS NULL THEN
        DBMS_OUTPUT.PUT_LINE('العمولة: بدون عمولة');
    ELSE
        DBMS_OUTPUT.PUT_LINE('العمولة: ' || (v_salary * v_comm));
    END IF;
END;
/

تمرين 8: تقرير موظف كامل

صعب

اكتب برنامج يطبع تقرير كامل عن موظف (رقم 100): الاسم الكامل، القسم، المرتب الشهري والسنوي، وعدد سنين الخبرة (من hire_date لحد النهاردة).

PL/SQL
DECLARE
    v_emp     employees%ROWTYPE;
    v_dept    departments.department_name%TYPE;
    v_years   NUMBER;
BEGIN
    SELECT * INTO v_emp
    FROM employees WHERE employee_id = 100;

    SELECT department_name INTO v_dept
    FROM departments WHERE department_id = v_emp.department_id;

    v_years := ROUND(MONTHS_BETWEEN(SYSDATE, v_emp.hire_date) / 12);

    DBMS_OUTPUT.PUT_LINE('========= تقرير موظف =========');
    DBMS_OUTPUT.PUT_LINE('الاسم: ' || v_emp.first_name || ' ' || v_emp.last_name);
    DBMS_OUTPUT.PUT_LINE('القسم: ' || v_dept);
    DBMS_OUTPUT.PUT_LINE('المرتب الشهري: ' || v_emp.salary);
    DBMS_OUTPUT.PUT_LINE('المرتب السنوي: ' || (v_emp.salary * 12));
    DBMS_OUTPUT.PUT_LINE('سنين الخبرة: ' || v_years);
    DBMS_OUTPUT.PUT_LINE('===============================');
END;

  • PL/SQL = Procedural Language / SQL (لغة برمجة بتشتغل جوه Oracle)
  • Block Structure: DECLARE → BEGIN → EXCEPTION → END
  • DBMS_OUTPUT.PUT_LINE للطباعة (لازم SET SERVEROUTPUT ON)
  • المتغيرات: v_name datatype := value; (علامة التعيين هي :=)
  • أنواع البيانات: VARCHAR2, NUMBER, DATE, BOOLEAN, INTEGER
  • %TYPE: المتغير ياخد نفس نوع العمود تلقائي
  • %ROWTYPE: متغير واحد يشيل صف كامل
  • الثوابت: CONSTANT (بتبدأ بـ c_)
  • SELECT INTO: جلب بيانات من الجدول في متغيرات
  • NULL + أي حاجة = NULL (استخدم NVL للحماية)
