إيه هو CREATE TABLE؟

تخيل إنك رايح مكتبة وعايز تعمل دفتر جديد لتسجيل بيانات الموظفين.

أول حاجة هتعملها إنك تحدد العناوين اللي في أول الدفتر:

📝 الاسم  |  📝 المرتب  |  📝 تاريخ التعيين  |  📝 القسم

أمر CREATE TABLE بيعمل نفس الحاجة بالظبط - بيعمل جدول جديد في قاعدة البيانات وبيحدد الأعمدة والأنواع بتاعتها.

CREATE TABLE = إنشاء جدول جديد في قاعدة البيانات مع تحديد أسماء الأعمدة ونوع البيانات في كل عمود
⌨️

الصيغة الأساسية (Syntax)

📐 القالب العام لـ CREATE TABLE
SQL Syntax
CREATE TABLE table_name ( column1 datatype [constraint], column2 datatype [constraint], column3 datatype [constraint], ... );
🔍 شرح كل جزء
الجزءالشرحمثال
CREATE TABLE الأمر نفسه - بيقول لـ Oracle: "اعمل جدول" ثابت - بتكتبه كل مرة
table_name اسم الجدول - انت اللي بتختاره employees
column اسم العمود first_name
datatype نوع البيانات - إيه نوع المعلومات في العمود ده VARCHAR2(50)
constraint قيود (اختياري) - شروط على البيانات NOT NULL

مثال بسيط - أول جدول

SQL - أول جدول
CREATE TABLE employees ( emp_id NUMBER, first_name VARCHAR2(50), salary NUMBER, hire_date DATE ); -- كده أنت عملت جدول اسمه employees -- فيه 4 أعمدة: رقم الموظف، الاسم، المرتب، تاريخ التعيين
💡
قواعد تسمية الجداول في Oracle:
• لازم يبدأ بحرف (مش رقم)
• ممكن يحتوي على حروف وأرقام و _ (underscore)
• مينفعش يكون اسم محجوز زي SELECT أو TABLE
• الحد الأقصى 128 حرف
• Oracle مش حساسة لحالة الحروف (employees = EMPLOYEES)

NUMBER - بالتفصيل

NUMBER هو نوع البيانات الرئيسي للأرقام في Oracle - بيخزن أرقام صحيحة وعشرية
📐 أشكال الكتابة
الشكلالمعنىمثال على القيمةاستخدام
NUMBER رقم بأي حجم (حتى 38 رقم) 123456789 لما مش عارف الحجم بالظبط
NUMBER(5) رقم صحيح - أقصى 5 أرقام 99999 رقم موظف، رقم قسم
NUMBER(8,2) 8 أرقام إجمالي، منهم 2 عشري 123456.78 مرتبات، أسعار، فلوس
NUMBER(3) رقم صحيح - أقصى 3 أرقام 999 عمر، نسبة مئوية
NUMBER(10,4) 10 أرقام إجمالي، 4 عشري 123456.7890 قياسات دقيقة
⚠️
فهم NUMBER(8,2):
الـ 8 = إجمالي عدد الأرقام (قبل وبعد العلامة العشرية)
الـ 2 = كم رقم بعد العلامة العشرية
يعني أقصى رقم: 999999.99 (6 أرقام قبل + 2 بعد = 8)
SQL - أمثلة NUMBER
-- أمثلة عملية لاستخدام NUMBER emp_id NUMBER(6) -- رقم موظف: 100001 salary NUMBER(8,2) -- مرتب: 15000.50 age NUMBER(3) -- عمر: 25 gpa NUMBER(3,2) -- تقدير: 3.75 phone NUMBER(11) -- رقم تليفون: 01012345678 price NUMBER(10,2) -- سعر: 1500000.00
📝

VARCHAR2 - بالتفصيل

VARCHAR2 هو نوع البيانات الأساسي للنصوص في Oracle - بيخزن نص بطول متغير (Variable Length)
📐 أشكال الكتابة
الشكلالمعنىمثالاستخدام
VARCHAR2(50) نص أقصى طوله 50 حرف 'أحمد محمود' أسماء
VARCHAR2(100) نص أقصى طوله 100 حرف 'ahmed@email.com' إيميلات
VARCHAR2(255) نص أقصى طوله 255 حرف 'شارع التحرير، القاهرة' عناوين
VARCHAR2(4000) نص طويل - أقصى حد! نص طويل جداً... ملاحظات، وصف
🤔 الفرق بين VARCHAR2 و CHAR
الموضوعVARCHAR2(10)CHAR(10)
لو خزّنت 'أحمد' (4 حروف) بيحجز 4 حروف بس ✅ بيحجز 10 حروف (ويملى الباقي فراغات) ❌
المساحة بيوفر مساحة ✅ بيضيع مساحة ❌
امتى تستخدمه أغلب الأوقات (الاختيار الافتراضي) بيانات ثابتة الطول (كود بلد: 'EG')
✅
القاعدة الذهبية: استخدم VARCHAR2 دايماً إلا لو البيانات طولها ثابت (زي كود البلد أو الجنس M/F).
SQL - أمثلة VARCHAR2
-- أمثلة عملية first_name VARCHAR2(50) -- اسم أول last_name VARCHAR2(50) -- اسم أخير email VARCHAR2(100) -- إيميل phone VARCHAR2(20) -- تليفون (نص مش رقم! عشان الـ 0 في الأول) address VARCHAR2(255) -- عنوان notes VARCHAR2(4000) -- ملاحظات gender CHAR(1) -- جنس: 'M' أو 'F' (ثابت حرف واحد) country_code CHAR(2) -- كود بلد: 'EG' (ثابت حرفين)
💡
ليه رقم التليفون VARCHAR2 مش NUMBER؟
لأن رقم التليفون بيبدأ بـ 0 (مثلاً 01012345678). لو حطيته NUMBER الصفر في الأول هيتشال! كمان مش هنعمل عليه عمليات حسابية.
📅

DATE - بالتفصيل

DATE بيخزن التاريخ والوقت مع بعض: السنة، الشهر، اليوم، الساعة، الدقيقة، الثانية
📐 صيغة DATE في Oracle
المعلومةالتفاصيل
يخزن إيهتاريخ + وقت (سنة، شهر، يوم، ساعة، دقيقة، ثانية)
الصيغة الافتراضيةDD-MON-YY مثال: '15-JAN-24'
النطاقمن 1 يناير 4712 قبل الميلاد إلى 31 ديسمبر 9999
الحجم7 bytes ثابت
SQL - التعامل مع DATE
-- إدخال تاريخ بالصيغة الافتراضية hire_date DATE → '15-JAN-24' -- إدخال تاريخ بصيغة محددة باستخدام TO_DATE TO_DATE('2024-01-15', 'YYYY-MM-DD') TO_DATE('15/01/2024', 'DD/MM/YYYY') TO_DATE('January 15, 2024', 'Month DD, YYYY') -- التاريخ والوقت الحالي SYSDATE -- بيرجع التاريخ والوقت دلوقتي -- عمليات على التواريخ SYSDATE + 7 -- بعد 7 أيام من دلوقتي SYSDATE - 30 -- قبل 30 يوم end_date - start_date -- الفرق بالأيام بين تاريخين
📆 رموز الصيغة (Format)
الرمزالمعنىمثال

MMالشهر (رقم)01

DDاليوم15

أنواع بيانات تانية مهمة

النوعالشرحمثالامتى تستخدمه
CLOB نص طويل جداً (حتى 4 GB!) مقالة كاملة لما VARCHAR2(4000) ميكفيش
BLOB بيانات ثنائية (صور، ملفات) صورة، PDF تخزين ملفات في الـ DB
TIMESTAMP تاريخ + وقت بدقة أعلى كسور الثانية لما محتاج دقة عالية
FLOAT رقم عشري (floating point) 3.14159 حسابات علمية
BOOLEAN صح أو غلط (Oracle 23c+) TRUE/FALSE حالات منطقية
💡
في 90% من الشغل هتستخدم 3 أنواع بس: NUMBER + VARCHAR2 + DATE. الباقي للحالات الخاصة.

القيود (Constraints)

القيود هي شروط وقوانين بتحطها على الأعمدة عشان تضمن إن البيانات صحيحة ومتكررش ومفيش حاجة ناقصة.

زي ما في الفورم بتاع أي موقع بيقولك "الإيميل مطلوب" أو "الاسم لازم يكون فريد".

PRIMARY KEY
المفتاح الأساسي - رقم فريد لكل صف (زي الرقم القومي). مينفعش يتكرر ومينفعش يكون فاضي.
emp_id NUMBER PRIMARY KEY
NOT NULL
العمود ده مينفعش يكون فاضي - لازم تحط قيمة فيه.
first_name VARCHAR2(50) NOT NULL
UNIQUE
القيمة لازم تكون فريدة - مينفعش صفين يكون عندهم نفس القيمة.
email VARCHAR2(100) UNIQUE
DEFAULT
قيمة افتراضية - لو مدخلتش قيمة هيحط القيمة دي تلقائياً.
status VARCHAR2(10) DEFAULT 'active'
CHECK
شرط معين - القيمة لازم تحقق شرط محدد.
age NUMBER CHECK (age >= 18)
FOREIGN KEY
مفتاح أجنبي - بيربط العمود ده بجدول تاني (العلاقات).
dept_id REFERENCES departments(dept_id)

مثال كامل عملي

هنعمل نظام بسيط فيه جدولين: الأقسام والموظفين

1. جدول الأقسام (departments)

SQL - إنشاء جدول الأقسام
CREATE TABLE departments ( dept_id NUMBER(4) PRIMARY KEY, dept_name VARCHAR2(50) NOT NULL UNIQUE, location VARCHAR2(100), created_at DATE DEFAULT SYSDATE ); -- dept_id: رقم القسم (فريد، أساسي) -- dept_name: اسم القسم (مطلوب، فريد) -- location: المكان (اختياري) -- created_at: تاريخ الإنشاء (تلقائياً تاريخ اليوم)

2. جدول الموظفين (employees)

SQL - إنشاء جدول الموظفين
CREATE TABLE employees ( emp_id NUMBER(6) PRIMARY KEY, first_name VARCHAR2(50) NOT NULL, last_name VARCHAR2(50) NOT NULL, email VARCHAR2(100) UNIQUE, phone VARCHAR2(20), salary NUMBER(8,2) CHECK (salary > 0), hire_date DATE DEFAULT SYSDATE, dept_id NUMBER(4) REFERENCES departments(dept_id) ); -- emp_id: رقم الموظف (فريد، أساسي) -- first_name, last_name: الاسم (مطلوب) -- email: إيميل (فريد - مينفعش اتنين بنفس الإيميل) -- phone: تليفون (اختياري) -- salary: المرتب (لازم يكون أكبر من 0) -- hire_date: تاريخ التعيين (تلقائياً اليوم) -- dept_id: رقم القسم (مرتبط بجدول departments)
📋 النتيجة: جدول الموظفين (فاضي - مستني بيانات)
EMP_ID 🔑 FIRST_NAME LAST_NAME EMAIL PHONE SALARY HIRE_DATE DEPT_ID 🔗
الجدول فاضي... هندخّل بيانات دلوقتي بـ INSERT INTO! 👇

INSERT INTO - إدخال البيانات

INSERT INTO = إضافة صف جديد (سجل جديد) في الجدول - زي ما بتملى صف جديد في Excel
📐 الصيغة الأساسية
SQL Syntax
-- الطريقة 1: تحديد الأعمدة (الأفضل ✅) INSERT INTO table_name (column1, column2, column3) VALUES (value1, value2, value3); -- الطريقة 2: بدون تحديد الأعمدة (لازم تحط كل القيم بالترتيب) INSERT INTO table_name VALUES (value1, value2, value3, ...);

مثال عملي - إدخال بيانات

SQL - إدخال أقسام
-- أولاً: ندخّل الأقسام (لازم الأقسام الأول عشان الموظفين مرتبطين بيها) INSERT INTO departments (dept_id, dept_name, location) VALUES (10, 'المبيعات', 'القاهرة'); INSERT INTO departments (dept_id, dept_name, location) VALUES (20, 'التسويق', 'الإسكندرية'); INSERT INTO departments (dept_id, dept_name, location) VALUES (30, 'الـ IT', 'القاهرة'); -- لاحظ: created_at مكتبناهاش - هتتحط تلقائياً (DEFAULT SYSDATE)
SQL - إدخال موظفين
-- ثانياً: ندخّل الموظفين INSERT INTO employees (emp_id, first_name, last_name, email, phone, salary, hire_date, dept_id) VALUES (1001, 'أحمد', 'محمود', 'ahmed@email.com', '01012345678', 8000, TO_DATE('2024-01-15', 'YYYY-MM-DD'), 10); INSERT INTO employees (emp_id, first_name, last_name, email, phone, salary, hire_date, dept_id) VALUES (1002, 'سارة', 'علي', 'sara@email.com', '01198765432', 9500.50, TO_DATE('2023-05-20', 'YYYY-MM-DD'), 20); INSERT INTO employees (emp_id, first_name, last_name, email, phone, salary, dept_id) VALUES (1003, 'محمد', 'سعيد', 'mohamed@email.com', '01555555555', 7000, 10); -- ^ لاحظ: hire_date مكتبناهاش - هتتحط تلقائياً تاريخ اليوم INSERT INTO employees (emp_id, first_name, last_name, email, salary, dept_id) VALUES (1004, 'نورا', 'حسن', 'noura@email.com', 11000, 30); -- ^ لاحظ: phone و hire_date مكتبناهمش - هيبقوا فاضيين (NULL) -- ⚠️ مهم: بعد ما تخلص إدخال البيانات، اكتب: COMMIT; -- ده بيحفظ البيانات نهائياً في الـ Database
📋 النتيجة: جدول الموظفين بعد الإدخال
EMP_ID FIRST_NAME LAST_NAME EMAIL SALARY DEPT_ID

⚠️
متنساش COMMIT!
في Oracle، البيانات مش بتتحفظ نهائياً غير لما تكتب COMMIT. لو قفلت البرنامج من غير COMMIT البيانات هتضيع! (أو اكتب ROLLBACK لو عايز تتراجع).

طرق إدخال بيانات متعددة

الطريقة 1: INSERT ALL (أكتر من صف مرة واحدة)

SQL - INSERT ALL
INSERT ALL INTO departments (dept_id, dept_name, location) VALUES (40, 'الموارد البشرية', 'القاهرة') INTO departments (dept_id, dept_name, location) VALUES (50, 'المحاسبة', 'الجيزة') INTO departments (dept_id, dept_name, location) VALUES (60, 'خدمة العملاء', 'المنصورة') SELECT * FROM DUAL; COMMIT;

الطريقة 2: INSERT مع NULL

SQL - إدخال مع قيم فارغة
-- لو عايز تحط قيمة فارغة صراحةً INSERT INTO employees (emp_id, first_name, last_name, email, phone, salary, dept_id) VALUES (1005, 'خالد', 'عمر', 'khaled@email.com', NULL, 6500, 20); -- ^^^^ -- phone = فاضي (NULL)

الطريقة 3: INSERT مع SYSDATE

SQL - إدخال مع تاريخ اليوم
-- تاريخ التعيين = النهاردة INSERT INTO employees (emp_id, first_name, last_name, email, salary, hire_date, dept_id) VALUES (1006, 'ياسمين', 'فوزي', 'yasmin@email.com', 8500, SYSDATE, 30);

أخطاء شائعة وإزاي تتجنبها

❌ الخطأ 1: تكرار PRIMARY KEY
INSERT INTO employees (emp_id, first_name, last_name, salary) VALUES (1001, 'علي', 'حسن', 5000); -- ❌ ERROR: 1001 موجود قبل كده! -- ORA-00001: unique constraint violated

✅ الحل: استخدم رقم emp_id مختلف (1007 مثلاً)

❌ الخطأ 2: نسيان قيمة NOT NULL
INSERT INTO employees (emp_id, email, salary) VALUES (1007, 'ali@email.com', 5000); -- ❌ ERROR: first_name و last_name مطلوبين (NOT NULL)! -- ORA-01400: cannot insert NULL

✅ الحل: حط قيمة لكل عمود NOT NULL

❌ الخطأ 3: FOREIGN KEY مش موجود
INSERT INTO employees (emp_id, first_name, last_name, salary, dept_id) VALUES (1007, 'علي', 'حسن', 5000, 99); -- ❌ ERROR: dept_id = 99 مش موجود في جدول departments! -- ORA-02291: integrity constraint violated - parent key not found

✅ الحل: استخدم dept_id موجود (10, 20, 30) أو ضيف القسم الأول

❌ الخطأ 4: نوع البيانات غلط
INSERT INTO employees (emp_id, first_name, last_name, salary) VALUES (1007, 'علي', 'حسن', 'كويس'); -- ❌ ERROR: salary عمود NUMBER ومينفعش تحط فيه نص! -- ORA-01722: invalid number

✅ الحل: حط رقم في الأعمدة الرقمية

❌ الخطأ 5: نسيان الـ Quotes للنصوص
INSERT INTO employees (emp_id, first_name, last_name, salary) VALUES (1007, علي, حسن, 5000); -- ❌ ERROR: النصوص لازم تكون بين ' ' (single quotes)

✅ الحل: 'علي' مش علي - دايماً حط النص بين علامتي اقتباس مفردة

SQL - حل التمرين 1
CREATE TABLE products ( product_id NUMBER(6) PRIMARY KEY, product_name VARCHAR2(100) NOT NULL, price NUMBER(10,2) CHECK (price > 0), category VARCHAR2(50), stock NUMBER DEFAULT 0 ); INSERT INTO products (product_id, product_name, price, category, stock) VALUES (1, 'لابتوب Dell', 15000.00, 'إلكترونيات', 25); INSERT INTO products (product_id, product_name, price, category, stock) VALUES (2, 'ماوس Logitech', 350.50, 'إلكترونيات', 100); INSERT INTO products (product_id, product_name, price, category, stock) VALUES (3, 'كرسي مكتب', 2500.00, 'أثاث', 15); INSERT INTO products (product_id, product_name, price, category) VALUES (4, 'سماعة Samsung', 1200.00, 'إلكترونيات'); -- stock هيبقى 0 تلقائياً (DEFAULT) INSERT INTO products (product_id, product_name, price, category, stock) VALUES (5, 'كتاب SQL', 180.00, 'كتب', 50); COMMIT; -- شوف البيانات: SELECT * FROM products;
📌 ملخص الدرس
  • ✓ CREATE TABLE = إنشاء جدول جديد
  • ✓ NUMBER(p,s) = أرقام (p أرقام، s عشري)
  • ✓ VARCHAR2(n) = نص متغير الطول
  • ✓ DATE = تاريخ ووقت
  • ✓ CHAR(n) = نص ثابت الطول
  • ✓ PRIMARY KEY = معرف فريد لكل صف
  • ✓ NOT NULL = لازم قيمة
  • ✓ UNIQUE = مينفعش يتكرر
  • ✓ CHECK = شرط على القيمة
  • ✓ DEFAULT = قيمة افتراضية
  • ✓ FOREIGN KEY = ربط بجدول تاني
  • ✓ INSERT INTO = إضافة بيانات
  • ✓ COMMIT = حفظ نهائي
  • ✓ TO_DATE = تحويل نص لتاريخ
