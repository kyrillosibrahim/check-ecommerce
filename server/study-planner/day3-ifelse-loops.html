الجزء الأول - الشروط

IF Statement

🚦

تشبيه: إشارة المرور!

لو الإشارة خضراء → امشي. لو حمراء → وقّف. كده بالظبط الـ IF بتشتغل: بتفحص شرط ولو متحقق بتنفذ كود معين.

الصيغة البسيطة

PL/SQL - IF Syntax 
IF condition THEN
    -- الكود ده هيتنفذ لو الشرط TRUE
    statements;
END IF;
PL/SQL - IF مثال عملي 
DECLARE
    v_salary NUMBER := 3500;
BEGIN
    IF v_salary < 5000 THEN
        DBMS_OUTPUT.PUT_LINE('المرتب تحت المتوسط - يحتاج زيادة!');
    END IF;

    DBMS_OUTPUT.PUT_LINE('تم الفحص');
END;
/
💡 ملاحظة: لازم تكتب END IF; (بمسافة!) مش ENDIF. ولازم الفاصلة المنقوطة ; بعدها.
شرط مزدوج

IF...ELSE

PL/SQL - IF...ELSE 
DECLARE
    v_grade NUMBER := 75;
BEGIN
    IF v_grade >= 50 THEN
        DBMS_OUTPUT.PUT_LINE('ناجح ✅');
    ELSE
        DBMS_OUTPUT.PUT_LINE('راسب ❌');
    END IF;
END;
/

مثال مع بيانات من الجدول

PL/SQL - IF...ELSE مع SELECT INTO 
DECLARE
    v_salary employees.salary%TYPE;
    v_name   employees.first_name%TYPE;
BEGIN
    SELECT first_name, salary
    INTO v_name, v_salary
    FROM employees
    WHERE employee_id = 100;

    DBMS_OUTPUT.PUT_LINE('الموظف: ' || v_name);

    IF v_salary > 10000 THEN
        DBMS_OUTPUT.PUT_LINE('مرتب عالي 💰');
    ELSE
        DBMS_OUTPUT.PUT_LINE('مرتب عادي');
    END IF;
END;
/
شروط متعددة

IF...ELSIF...ELSE

لما يكون عندك أكتر من شرطين:

PL/SQL - IF...ELSIF...ELSE 
DECLARE
    v_salary NUMBER := 7500;
    v_level  VARCHAR2(20);
BEGIN
    IF v_salary >= 15000 THEN
        v_level := 'Senior';
    ELSIF v_salary >= 10000 THEN
        v_level := 'Mid-Level';
    ELSIF v_salary >= 5000 THEN
        v_level := 'Junior';
    ELSE
        v_level := 'Trainee';
    END IF;

    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('المستوى: ' || v_level);
END;
/
⚠️ لاحظ: الكلمة هي ELSIF مش ELSEIF ولا ELSE IF! دي من أشهر الأخطاء عند المبتدئين.

IF مع عمليات منطقية (AND, OR, NOT)

PL/SQL - Logical Operators 
DECLARE
    v_age     NUMBER := 28;
    v_salary  NUMBER := 8000;
    v_dept_id NUMBER := 20;
BEGIN
    -- AND: الشرطين لازم يتحققوا
    IF v_age > 25 AND v_salary > 7000 THEN
        DBMS_OUTPUT.PUT_LINE('مؤهل للترقية ✅');
    END IF;

    -- OR: واحد من الشرطين يكفي
    IF v_dept_id = 10 OR v_dept_id = 20 THEN
        DBMS_OUTPUT.PUT_LINE('في قسم رئيسي');
    END IF;

    -- NOT: عكس الشرط
    IF NOT (v_salary < 5000) THEN
        DBMS_OUTPUT.PUT_LINE('المرتب فوق الحد الأدنى');
    END IF;
END;
/

CASE Statement

لو عندك شروط كتير على نفس المتغير، CASE أنظف وأوضح من ELSIF:

Simple CASE

PL/SQL - Simple CASE 
DECLARE
    v_dept_id NUMBER := 20;
    v_dept_name VARCHAR2(50);
BEGIN
    CASE v_dept_id
        WHEN 10 THEN v_dept_name := 'Administration';
        WHEN 20 THEN v_dept_name := 'Marketing';
        WHEN 30 THEN v_dept_name := 'IT';
        WHEN 40 THEN v_dept_name := 'HR';
        ELSE v_dept_name := 'Unknown';
    END CASE;

    DBMS_OUTPUT.PUT_LINE('القسم: ' || v_dept_name);
END;
/

Searched CASE (مع شروط مختلفة)

PL/SQL - Searched CASE 
DECLARE
    v_salary NUMBER := 12000;
    v_bonus  NUMBER;
BEGIN
    CASE
        WHEN v_salary >= 15000 THEN v_bonus := v_salary * 0.20;
        WHEN v_salary >= 10000 THEN v_bonus := v_salary * 0.15;
        WHEN v_salary >= 5000  THEN v_bonus := v_salary * 0.10;
        ELSE v_bonus := v_salary * 0.05;
    END CASE;

    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_salary);
    DBMS_OUTPUT.PUT_LINE('البونص: ' || v_bonus);
END;
/
الجزء الثاني - التكرار

Basic LOOP

🔁

تشبيه: الـ Loop زي الغسالة!

بتفضل تدور لحد ما تنتهي الدورة (الشرط يتحقق). لو منسيتش تحط شرط الإيقاف، هتفضل تدور للأبد! (infinite loop)

PL/SQL - Basic LOOP 
DECLARE
    v_counter NUMBER := 1;
BEGIN
    LOOP
        DBMS_OUTPUT.PUT_LINE('العدد: ' || v_counter);
        v_counter := v_counter + 1;

        EXIT WHEN v_counter > 5;  -- شرط الإيقاف!
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('انتهى!');
END;
/

النتيجة:

Output
العدد: 1
العدد: 2
العدد: 3
العدد: 4
العدد: 5
انتهى!
🚨 تحذير: لو نسيت EXIT WHEN هيكون عندك infinite loop! دايماً تأكد إن فيه شرط إيقاف.
تكرار بشرط

WHILE LOOP

بتفحص الشرط قبل كل دورة. لو الشرط FALSE من البداية، مش هينفذ حاجة:

PL/SQL - WHILE LOOP 
DECLARE
    v_counter NUMBER := 1;
    v_sum     NUMBER := 0;
BEGIN
    WHILE v_counter <= 10 LOOP
        v_sum := v_sum + v_counter;
        v_counter := v_counter + 1;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('مجموع الأرقام من 1 لـ 10 = ' || v_sum);
    -- Output: مجموع الأرقام من 1 لـ 10 = 55
END;
/

مثال عملي: مضاعفة المرتب

PL/SQL - WHILE مثال 
DECLARE
    v_salary NUMBER := 3000;
    v_years  NUMBER := 0;
BEGIN
    -- كل سنة زيادة 10% - كام سنة عشان يوصل 10000؟
    WHILE v_salary < 10000 LOOP
        v_salary := v_salary * 1.10;
        v_years := v_years + 1;
        DBMS_OUTPUT.PUT_LINE('سنة ' || v_years || ': ' || ROUND(v_salary));
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('هيوصل 10000 بعد ' || v_years || ' سنة');
END;
/
الأكثر استخداماً

FOR LOOP

الـ FOR LOOP هو الأسهل والأكثر استخداماً! مش محتاج تعرّف counter ولا تزوّده - كل ده بيحصل تلقائي:

PL/SQL - FOR LOOP 
BEGIN
    -- من 1 لـ 5
    FOR i IN 1..5 LOOP
        DBMS_OUTPUT.PUT_LINE('i = ' || i);
    END LOOP;
END;
/

-- Output:
-- i = 1
-- i = 2
-- i = 3
-- i = 4
-- i = 5

FOR LOOP بالعكس (REVERSE)

PL/SQL - FOR REVERSE 
BEGIN
    -- العد التنازلي!
    FOR i IN REVERSE 1..5 LOOP
        DBMS_OUTPUT.PUT_LINE(i);
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('🚀 انطلاق!');
END;
/

-- Output: 5, 4, 3, 2, 1, 🚀 انطلاق!

مثال عملي: جدول الضرب

PL/SQL - جدول الضرب 
DECLARE
    v_num NUMBER := 7;
BEGIN
    DBMS_OUTPUT.PUT_LINE('جدول ضرب ' || v_num);
    DBMS_OUTPUT.PUT_LINE('-------------------');

    FOR i IN 1..12 LOOP
        DBMS_OUTPUT.PUT_LINE(v_num || ' × ' || i || ' = ' || (v_num * i));
    END LOOP;
END;
/
✅ ميزة FOR LOOP:
  • مش محتاج DECLARE للـ counter (i بيتعرف تلقائي)
  • مش محتاج تزوّد الـ counter (بيزيد لوحده)
  • مفيش خطر infinite loop (العدد محدد)
متقدم

Nested Loops + Labels

PL/SQL - Nested FOR Loops 
BEGIN
    -- طباعة مستطيل من نجوم 5×3
    <<outer_loop>>
    FOR row_num IN 1..3 LOOP
        FOR col_num IN 1..5 LOOP
            DBMS_OUTPUT.PUT('* ');
        END LOOP;
        DBMS_OUTPUT.NEW_LINE;
    END LOOP outer_loop;
END;
/

-- Output:
-- * * * * *
-- * * * * *
-- * * * * *
💡 Labels: الـ <<outer_loop>> دي label بتسميها. مفيدة لما يكون عندك loops جوه بعض وعايز تعمل EXIT للـ loop الخارجي: EXIT outer_loop WHEN ...
مقارنة

مقارنة الـ Loops الثلاثة

النوع الفحص استخدمها لما... مثال
Basic LOOP بعد التنفيذ عايز ينفذ مرة على الأقل LOOP...EXIT WHEN
WHILE LOOP قبل التنفيذ مش عارف هينفذ كام مرة WHILE x < 100 LOOP
FOR LOOP عدد محدد عارف عدد التكرارات FOR i IN 1..10 LOOP
نصيحة: في 90% من الحالات هتستخدم FOR LOOP. استخدم WHILE لما يكون الشرط ديناميكي. واستخدم Basic LOOP في حالات نادرة (مثلاً لما تقرأ من cursor).

Procedure مع Parameter

PL/SQL - Procedure مع Input 
CREATE OR REPLACE PROCEDURE greet_employee(
    p_emp_id IN NUMBER
)
IS
    v_name employees.first_name%TYPE;
BEGIN
    SELECT first_name INTO v_name
    FROM employees
    WHERE employee_id = p_emp_id;

    DBMS_OUTPUT.PUT_LINE('مرحباً يا ' || v_name || '! 👋');
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('الموظف رقم ' || p_emp_id || ' مش موجود!');
END;
/

-- استخدام:
BEGIN
    greet_employee(100);  -- مرحباً يا Steven! 👋
    greet_employee(999);  -- الموظف رقم 999 مش موجود!
END;
/
نصيحة: استخدم p_ كبداية لأسماء الـ Parameters عشان تفرّقها عن المتغيرات v_ والثوابت c_.
Parameters

أنواع الـ Parameters: IN, OUT, IN OUT

النوع الاتجاه الوصف Default?
IN داخل ← بتبعت قيمة للـ Procedure (للقراءة بس) ✅ نعم (الـ Default)
OUT → خارج الـ Procedure بتبعت قيمة للي ناداها ❌
IN OUT ← → الاتنين بتستقبل وبتبعت ❌

مثال: OUT Parameter

PL/SQL - OUT Parameter 
CREATE OR REPLACE PROCEDURE get_employee_info(
    p_emp_id    IN  NUMBER,
    p_name      OUT VARCHAR2,
    p_salary    OUT NUMBER,
    p_dept_name OUT VARCHAR2
)
IS
BEGIN
    SELECT e.first_name || ' ' || e.last_name,
           e.salary,
           d.department_name
    INTO p_name, p_salary, p_dept_name
    FROM employees e
    JOIN departments d ON e.department_id = d.department_id
    WHERE e.employee_id = p_emp_id;
END;
/

-- استخدام:
DECLARE
    v_name VARCHAR2(100);
    v_sal  NUMBER;
    v_dept VARCHAR2(50);
BEGIN
    get_employee_info(100, v_name, v_sal, v_dept);

    DBMS_OUTPUT.PUT_LINE('الاسم: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_sal);
    DBMS_OUTPUT.PUT_LINE('القسم: ' || v_dept);
END;
/

مثال: IN OUT Parameter

PL/SQL - IN OUT Parameter 
-- Procedure بتاخد مرتب وتحوّله لمرتب بعد الزيادة
CREATE OR REPLACE PROCEDURE apply_raise(
    p_salary    IN OUT NUMBER,
    p_percentage IN     NUMBER DEFAULT 10  -- قيمة افتراضية 10%
)
IS
BEGIN
    p_salary := p_salary + (p_salary * p_percentage / 100);
END;
/

-- استخدام:
DECLARE
    v_salary NUMBER := 5000;
BEGIN
    DBMS_OUTPUT.PUT_LINE('قبل: ' || v_salary);   -- 5000

    apply_raise(v_salary);         -- 10% default
    DBMS_OUTPUT.PUT_LINE('بعد 10%: ' || v_salary); -- 5500

    apply_raise(v_salary, 20);     -- 20% زيادة
    DBMS_OUTPUT.PUT_LINE('بعد 20%: ' || v_salary); -- 6600
END;
/
💡 DEFAULT: تقدر تحط قيمة افتراضية للـ IN Parameters. كده لو مبعتش قيمة، هياخد الـ Default.
أمثلة عملية

أمثلة متقدمة على Procedures

Procedure: زيادة مرتب موظف في الجدول

PL/SQL - UPDATE في Procedure 
CREATE OR REPLACE PROCEDURE raise_salary(
    p_emp_id     IN NUMBER,
    p_percentage IN NUMBER
)
IS
    v_old_salary employees.salary%TYPE;
    v_new_salary employees.salary%TYPE;
BEGIN
    -- جيب المرتب الحالي
    SELECT salary INTO v_old_salary
    FROM employees WHERE employee_id = p_emp_id;

    -- احسب المرتب الجديد
    v_new_salary := v_old_salary + (v_old_salary * p_percentage / 100);

    -- حدّث الجدول
    UPDATE employees
    SET salary = v_new_salary
    WHERE employee_id = p_emp_id;

    DBMS_OUTPUT.PUT_LINE('الموظف ' || p_emp_id || ':');
    DBMS_OUTPUT.PUT_LINE('  المرتب القديم: ' || v_old_salary);
    DBMS_OUTPUT.PUT_LINE('  المرتب الجديد: ' || v_new_salary);

    COMMIT;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('خطأ: الموظف ' || p_emp_id || ' مش موجود!');
END;
/

-- استخدام: زوّد مرتب الموظف 100 بنسبة 15%
EXEC raise_salary(100, 15);

Procedure: تقرير قسم كامل

PL/SQL - Procedure مع IF + Loop 
CREATE OR REPLACE PROCEDURE dept_report(
    p_dept_id IN NUMBER
)
IS
    v_dept_name departments.department_name%TYPE;
    v_count     NUMBER;
    v_avg_sal   NUMBER;
    v_max_sal   NUMBER;
    v_min_sal   NUMBER;
    v_status    VARCHAR2(20);
BEGIN
    -- اسم القسم
    SELECT department_name INTO v_dept_name
    FROM departments WHERE department_id = p_dept_id;

    -- الإحصائيات
    SELECT COUNT(*), ROUND(AVG(salary)), MAX(salary), MIN(salary)
    INTO v_count, v_avg_sal, v_max_sal, v_min_sal
    FROM employees WHERE department_id = p_dept_id;

    -- تحديد الحالة
    IF v_count > 10 THEN
        v_status := 'قسم كبير';
    ELSIF v_count > 5 THEN
        v_status := 'قسم متوسط';
    ELSE
        v_status := 'قسم صغير';
    END IF;

    -- الطباعة
    DBMS_OUTPUT.PUT_LINE('╔══════════════════════════╗');
    DBMS_OUTPUT.PUT_LINE('  تقرير قسم: ' || v_dept_name);
    DBMS_OUTPUT.PUT_LINE('╠══════════════════════════╣');
    DBMS_OUTPUT.PUT_LINE('  عدد الموظفين: ' || v_count);
    DBMS_OUTPUT.PUT_LINE('  متوسط المرتب: ' || v_avg_sal);
    DBMS_OUTPUT.PUT_LINE('  أعلى مرتب: ' || v_max_sal);
    DBMS_OUTPUT.PUT_LINE('  أقل مرتب: ' || v_min_sal);
    DBMS_OUTPUT.PUT_LINE('  الحالة: ' || v_status);
    DBMS_OUTPUT.PUT_LINE('╚══════════════════════════╝');
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('القسم مش موجود!');
END;
/

EXEC dept_report(20);

تمارين عملية

تمرين 1: تصنيف الدرجات

سهل

اكتب برنامج يأخذ درجة (v_grade = 85) ويطبع التقدير: A (90+)، B (80+)، C (70+)، D (60+)، F (أقل).

PL/SQL
DECLARE
    v_grade NUMBER := 85;
    v_rating VARCHAR2(10);
BEGIN
    IF v_grade >= 90 THEN v_rating := 'A - ممتاز';
    ELSIF v_grade >= 80 THEN v_rating := 'B - جيد جداً';
    ELSIF v_grade >= 70 THEN v_rating := 'C - جيد';
    ELSIF v_grade >= 60 THEN v_rating := 'D - مقبول';
    ELSE v_rating := 'F - راسب';
    END IF;

    DBMS_OUTPUT.PUT_LINE('الدرجة: ' || v_grade);
    DBMS_OUTPUT.PUT_LINE('التقدير: ' || v_rating);
END;
/

تمرين 2: CASE مع أيام الأسبوع

سهل

اكتب برنامج يأخذ رقم اليوم (1-7) ويطبع اسمه بالعربي واللو كان عطلة أو شغل.

PL/SQL
DECLARE
    v_day_num NUMBER := 6;
    v_day_name VARCHAR2(20);
    v_type VARCHAR2(10);
BEGIN
    CASE v_day_num
        WHEN 1 THEN v_day_name := 'الأحد';
        WHEN 2 THEN v_day_name := 'الاثنين';
        WHEN 3 THEN v_day_name := 'الثلاثاء';
        WHEN 4 THEN v_day_name := 'الأربعاء';
        WHEN 5 THEN v_day_name := 'الخميس';
        WHEN 6 THEN v_day_name := 'الجمعة';
        WHEN 7 THEN v_day_name := 'السبت';
    END CASE;

    IF v_day_num IN (6, 7) THEN
        v_type := 'عطلة 🎉';
    ELSE
        v_type := 'شغل 💼';
    END IF;

    DBMS_OUTPUT.PUT_LINE(v_day_name || ' - ' || v_type);
END;
/

تمرين 3: Factorial بالـ FOR

متوسط

اكتب برنامج يحسب الـ Factorial لعدد معين (مثلاً 5! = 5×4×3×2×1 = 120) باستخدام FOR LOOP.

PL/SQL
DECLARE
    v_num    NUMBER := 5;
    v_result NUMBER := 1;
BEGIN
    FOR i IN 1..v_num LOOP
        v_result := v_result * i;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE(v_num || '! = ' || v_result);
    -- Output: 5! = 120
END;
/

تمرين 4: جدول الضرب بـ WHILE

متوسط

اكتب برنامج يطبع جدول ضرب الـ 9 من 1 لـ 12 باستخدام WHILE LOOP.

PL/SQL
DECLARE
    v_num NUMBER := 9;
    v_i   NUMBER := 1;
BEGIN
    DBMS_OUTPUT.PUT_LINE('جدول ضرب ' || v_num);
    WHILE v_i <= 12 LOOP
        DBMS_OUTPUT.PUT_LINE(v_num || ' × ' || v_i || ' = ' || (v_num * v_i));
        v_i := v_i + 1;
    END LOOP;
END;
/

تمرين 5: Fizz Buzz

متوسط

اكتب برنامج يطبع الأرقام من 1 لـ 30. لو العدد يقبل القسمة على 3 اطبع "Fizz"، لو على 5 اطبع "Buzz"، لو على 3 و 5 اطبع "FizzBuzz"، غير كده اطبع العدد.

PL/SQL
BEGIN
    FOR i IN 1..30 LOOP
        IF MOD(i, 15) = 0 THEN
            DBMS_OUTPUT.PUT_LINE(i || ' - FizzBuzz');
        ELSIF MOD(i, 3) = 0 THEN
            DBMS_OUTPUT.PUT_LINE(i || ' - Fizz');
        ELSIF MOD(i, 5) = 0 THEN
            DBMS_OUTPUT.PUT_LINE(i || ' - Buzz');
        ELSE
            DBMS_OUTPUT.PUT_LINE(i);
        END IF;
    END LOOP;
END;
/

تمرين 6: أول Procedure بتاعك

متوسط

اعمل Procedure اسمها print_stars بتاخد عدد (n) وبتطبع n نجمة في سطر واحد. مثلاً print_stars(5) بتطبع: *****

PL/SQL
CREATE OR REPLACE PROCEDURE print_stars(
    p_count IN NUMBER
)
IS
    v_line VARCHAR2(200) := '';
BEGIN
    FOR i IN 1..p_count LOOP
        v_line := v_line || '*';
    END LOOP;
    DBMS_OUTPUT.PUT_LINE(v_line);
END;
/

BEGIN
    print_stars(5);   -- *****
    print_stars(10);  -- **********
    print_stars(3);   -- ***
END;
/

تمرين 7: Procedure مع OUT

صعب

اعمل Procedure اسمها calc_tax بتاخد المرتب (IN) وبتطلع: الضريبة (OUT) والصافي (OUT). الضريبة 14%.

PL/SQL
CREATE OR REPLACE PROCEDURE calc_tax(
    p_salary IN  NUMBER,
    p_tax    OUT NUMBER,
    p_net    OUT NUMBER
)
IS
    c_tax_rate CONSTANT NUMBER := 0.14;
BEGIN
    p_tax := ROUND(p_salary * c_tax_rate, 2);
    p_net := p_salary - p_tax;
END;
/

DECLARE
    v_tax NUMBER;
    v_net NUMBER;
BEGIN
    calc_tax(8000, v_tax, v_net);
    DBMS_OUTPUT.PUT_LINE('الضريبة: ' || v_tax);  -- 1120
    DBMS_OUTPUT.PUT_LINE('الصافي: ' || v_net);   -- 6880
END;
/

تمرين 8: Procedure لزيادة المرتبات

صعب

اعمل Procedure اسمها raise_dept_salary بتأخد department_id ونسبة زيادة. بتزوّد مرتب كل موظفين القسم. بتطبع عدد الموظفين اللي اتأثروا.

PL/SQL
CREATE OR REPLACE PROCEDURE raise_dept_salary(
    p_dept_id    IN NUMBER,
    p_percentage IN NUMBER
)
IS
    v_count NUMBER;
BEGIN
    UPDATE employees
    SET salary = salary + (salary * p_percentage / 100)
    WHERE department_id = p_dept_id;

    v_count := SQL%ROWCOUNT;  -- عدد الصفوف اللي اتأثرت

    IF v_count = 0 THEN
        DBMS_OUTPUT.PUT_LINE('مفيش موظفين في القسم ' || p_dept_id);
    ELSE
        DBMS_OUTPUT.PUT_LINE('تم زيادة ' || v_count ||
            ' موظف بنسبة ' || p_percentage || '%');
        COMMIT;
    END IF;
END;
/

EXEC raise_dept_salary(20, 10);

تمرين 9: هرم النجوم

صعب

اعمل Procedure اسمها print_pyramid بتاخد عدد الصفوف وبتطبع هرم. مثلاً print_pyramid(4):

   *
  ***
 *****
*******

كل صف فيه (2*i - 1) نجمة و(n - i) مسافة قبلها. استخدم RPAD أو loop داخلي.

PL/SQL
CREATE OR REPLACE PROCEDURE print_pyramid(
    p_rows IN NUMBER
)
IS
    v_line VARCHAR2(200);
BEGIN
    FOR i IN 1..p_rows LOOP
        v_line := LPAD(' ', p_rows - i) ||
                 RPAD('*', 2 * i - 1, '*');
        DBMS_OUTPUT.PUT_LINE(v_line);
    END LOOP;
END;
/

EXEC print_pyramid(5);

تمرين 10: سيناريو متكامل

تحدي

اعمل Procedure اسمها employee_report بتاخد employee_id وبتطبع:

  • اسم الموظف والقسم والمرتب
  • التصنيف (Senior لو > 15000، Mid لو > 8000، Junior غير كده)
  • المرتب السنوي
  • عدد سنين الخبرة
  • لو فيه عمولة اطبعها، غير كده اطبع "بدون عمولة"
PL/SQL
CREATE OR REPLACE PROCEDURE employee_report(
    p_emp_id IN NUMBER
)
IS
    v_emp     employees%ROWTYPE;
    v_dept    departments.department_name%TYPE;
    v_level   VARCHAR2(20);
    v_years   NUMBER;
BEGIN
    SELECT * INTO v_emp
    FROM employees WHERE employee_id = p_emp_id;

    SELECT department_name INTO v_dept
    FROM departments WHERE department_id = v_emp.department_id;

    -- التصنيف
    CASE
        WHEN v_emp.salary > 15000 THEN v_level := 'Senior';
        WHEN v_emp.salary > 8000  THEN v_level := 'Mid-Level';
        ELSE v_level := 'Junior';
    END CASE;

    v_years := ROUND(MONTHS_BETWEEN(SYSDATE, v_emp.hire_date) / 12);

    DBMS_OUTPUT.PUT_LINE('╔═══════════════════════════════╗');
    DBMS_OUTPUT.PUT_LINE('        تقرير موظف #' || p_emp_id);
    DBMS_OUTPUT.PUT_LINE('╠═══════════════════════════════╣');
    DBMS_OUTPUT.PUT_LINE('  الاسم: ' || v_emp.first_name || ' ' || v_emp.last_name);
    DBMS_OUTPUT.PUT_LINE('  القسم: ' || v_dept);
    DBMS_OUTPUT.PUT_LINE('  المستوى: ' || v_level);
    DBMS_OUTPUT.PUT_LINE('  المرتب الشهري: ' || v_emp.salary);
    DBMS_OUTPUT.PUT_LINE('  المرتب السنوي: ' || (v_emp.salary * 12));
    DBMS_OUTPUT.PUT_LINE('  سنين الخبرة: ' || v_years);

    IF v_emp.commission_pct IS NOT NULL THEN
        DBMS_OUTPUT.PUT_LINE('  العمولة: ' ||
            ROUND(v_emp.salary * v_emp.commission_pct));
    ELSE
        DBMS_OUTPUT.PUT_LINE('  العمولة: بدون عمولة');
    END IF;

    DBMS_OUTPUT.PUT_LINE('╚═══════════════════════════════╝');

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('الموظف #' || p_emp_id || ' مش موجود!');
END;
/

EXEC employee_report(100);

ملخص اللي اتعلمناه

  • IF...THEN...END IF للشروط البسيطة
  • IF...ELSIF...ELSE للشروط المتعددة (لاحظ: ELSIF مش ELSEIF!)
  • CASE Statement بديل أنظف لما الشروط كتير على نفس المتغير
  • AND, OR, NOT للعمليات المنطقية
  • Basic LOOP + EXIT WHEN (بتنفذ مرة على الأقل)
  • WHILE LOOP (بتفحص الشرط قبل كل دورة)
  • FOR LOOP (الأسهل - counter تلقائي + REVERSE)
  • Stored Procedure = كود مخزن في الداتابيز بتنادي عليه بالاسم
  • CREATE OR REPLACE PROCEDURE (IS بدل DECLARE)
  • Parameters: IN (الـ default)، OUT (إرجاع قيمة)، IN OUT (الاتنين)
  • p_ لأسماء الـ Parameters، v_ للمتغيرات، c_ للثوابت
  • SQL%ROWCOUNT لعدد الصفوف اللي اتأثرت
