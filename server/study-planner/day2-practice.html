المستوى السهل

سهل تمرين 1: إنشاء قاعدة البيانات ◀
أنشئ 4 جداول للشركة:
1. departments: dept_id (PK), dept_name, location, budget
2. employees: emp_id (PK), first_name, last_name, email (UNIQUE), salary, hire_date, dept_id (FK), manager_id
3. projects: project_id (PK), project_name, dept_id (FK), start_date, budget
4. customers: customer_id (PK), customer_name, city, contact_email, assigned_emp_id (FK)
ابدأ بالجدول اللي مالوش FK أول (departments)، بعدين employees، بعدين projects و customers.
CREATE TABLE departments ( dept_id NUMBER(4) PRIMARY KEY, dept_name VARCHAR2(50) NOT NULL, location VARCHAR2(50), budget NUMBER(12,2) ); CREATE TABLE employees ( emp_id NUMBER(6) PRIMARY KEY, first_name VARCHAR2(30) NOT NULL, last_name VARCHAR2(30) NOT NULL, email VARCHAR2(60) UNIQUE, salary NUMBER(10,2), hire_date DATE, dept_id NUMBER(4) REFERENCES departments(dept_id), manager_id NUMBER(6) ); CREATE TABLE projects ( project_id NUMBER(4) PRIMARY KEY, project_name VARCHAR2(60) NOT NULL, dept_id NUMBER(4) REFERENCES departments(dept_id), start_date DATE, budget NUMBER(12,2) ); CREATE TABLE customers ( customer_id NUMBER(6) PRIMARY KEY, customer_name VARCHAR2(60) NOT NULL, city VARCHAR2(50), contact_email VARCHAR2(60), assigned_emp_id NUMBER(6) REFERENCES employees(emp_id) );

سهل تمرين 3: UPDATE و DELETE ◀
1. زوّد مرتب كل موظفين قسم التقنية (20) بنسبة 10%
2. عيّن منى (108) لقسم التسويق (40)
3. عيّن هدى (112) لقسم المالية (50)
4. حدّث مرتب ريم (110) لـ 6500
استخدم UPDATE مع SET و WHERE. للنسبة: salary = salary * 1.10
-- 1. زيادة 10% لقسم التقنية UPDATE employees SET salary = salary * 1.10 WHERE dept_id = 20; -- 2. تعيين منى لقسم التسويق UPDATE employees SET dept_id = 40 WHERE emp_id = 108; -- 3. تعيين هدى لقسم المالية UPDATE employees SET dept_id = 50 WHERE emp_id = 112; -- 4. تحديث مرتب ريم UPDATE employees SET salary = 6500 WHERE emp_id = 110; COMMIT;
سهل تمرين 4: إحصائيات أساسية ◀
اعرض في Query واحد: عدد الموظفين الكلي، عدد اللي عندهم مرتب، إجمالي المرتبات، المتوسط (مقرب)، أعلى مرتب، أقل مرتب.
استخدم COUNT(*), COUNT(salary), SUM, ROUND(AVG()), MAX, MIN كلهم في SELECT واحد.
SELECT COUNT(*) AS total_emp, COUNT(salary) AS with_salary, SUM(salary) AS total_sal, ROUND(AVG(salary)) AS avg_sal, MAX(salary) AS max_sal, MIN(salary) AS min_sal FROM employees;

المستوى المتوسط

متوسط تمرين 5: GROUP BY + HAVING◀
اعرض لكل قسم: عدد الموظفين، إجمالي المرتبات، متوسط المرتب. بس الأقسام اللي فيها أكتر من موظف واحد ومتوسط مرتبها فوق 8000. رتب تنازلياً حسب المتوسط.
GROUP BY dept_id ثم HAVING مع شرطين: COUNT(*) > 1 AND AVG(salary) > 8000
SELECT dept_id, COUNT(*) AS emp_count, SUM(salary) AS total_sal, ROUND(AVG(salary)) AS avg_sal FROM employees GROUP BY dept_id HAVING COUNT(*) > 1 AND AVG(salary) > 8000 ORDER BY avg_sal DESC;
متوسط تمرين 6: INNER JOIN - التقرير الأساسي◀
اعرض اسم كل موظف واسم قسمه ومكان القسم ومرتبه. بس الموظفين اللي في القاهرة ومرتبهم فوق 10000. رتب حسب المرتب تنازلياً.
INNER JOIN employees مع departments على dept_id. WHERE على location و salary.
SELECT e.first_name, e.last_name, d.dept_name, d.location, e.salary FROM employees e INNER JOIN departments d ON e.dept_id = d.dept_id WHERE d.location = 'القاهرة' AND e.salary > 10000 ORDER BY e.salary DESC;
متوسط تمرين 7: LEFT JOIN - المفقودين◀
1. اعرض العملاء اللي مالهمش موظف مسؤول عنهم
2. اعرض الموظفين اللي مش مسؤولين عن أي عميل
LEFT JOIN + WHERE ... IS NULL. للسؤال الأول: customers LEFT JOIN employees. للتاني: employees LEFT JOIN customers.
-- 1. عملاء بدون موظف مسؤول SELECT c.customer_name, c.city FROM customers c LEFT JOIN employees e ON c.assigned_emp_id = e.emp_id WHERE e.emp_id IS NULL; -- 2. موظفين مش مسؤولين عن أي عميل SELECT e.first_name, e.last_name, d.dept_name FROM employees e LEFT JOIN customers c ON e.emp_id = c.assigned_emp_id LEFT JOIN departments d ON e.dept_id = d.dept_id WHERE c.customer_id IS NULL;
متوسط تمرين 8: JOIN + GROUP BY - تقرير الأقسام◀
اعرض اسم كل قسم وعدد موظفينه ومتوسط مرتبه وأعلى مرتب. حتى الأقسام الفاضية تظهر. رتب حسب عدد الموظفين تنازلياً.
departments LEFT JOIN employees. GROUP BY dept_name. استخدم COUNT(e.emp_id) مش COUNT(*) عشان الأقسام الفاضية تعرض 0.
SELECT d.dept_name, COUNT(e.emp_id) AS emp_count, ROUND(AVG(e.salary)) AS avg_salary, MAX(e.salary) AS max_salary FROM departments d LEFT JOIN employees e ON d.dept_id = e.dept_id GROUP BY d.dept_name ORDER BY emp_count DESC;

المستوى الصعب

صعب تمرين 9: Multi JOIN - 3 جداول◀
اعرض اسم كل مشروع واسم القسم بتاعه وعدد الموظفين في القسم ده وميزانية المشروع. رتب حسب الميزانية تنازلياً.
projects JOIN departments JOIN employees. استخدم COUNT(DISTINCT e.emp_id) عشان لو القسم عنده أكتر من مشروع الموظف ميتعدش مرتين.
SELECT p.project_name, d.dept_name, COUNT(DISTINCT e.emp_id) AS emp_count, p.budget FROM projects p INNER JOIN departments d ON p.dept_id = d.dept_id LEFT JOIN employees e ON d.dept_id = e.dept_id GROUP BY p.project_name, d.dept_name, p.budget ORDER BY p.budget DESC;
صعب تمرين 10: SELF JOIN - الهيكل الإداري◀
1. اعرض كل موظف مع اسم مديره (اعرض "المدير العام" للي مالوش مدير)
2. اعرض الموظفين اللي مرتبهم أعلى من مرتب مديرهم وفرق المرتب
LEFT JOIN employees مع نفسها: ON e.manager_id = m.emp_id. استخدم NVL للـ NULL.
-- 1. كل موظف مع مديره SELECT e.first_name || ' ' || e.last_name AS employee, e.salary, NVL(m.first_name || ' ' || m.last_name, 'المدير العام') AS manager FROM employees e LEFT JOIN employees m ON e.manager_id = m.emp_id ORDER BY e.emp_id; -- 2. الموظفين الأغنى من مديرهم SELECT e.first_name AS employee, e.salary AS emp_salary, m.first_name AS manager, m.salary AS mgr_salary, e.salary - m.salary AS difference FROM employees e INNER JOIN employees m ON e.manager_id = m.emp_id WHERE e.salary > m.salary ORDER BY difference DESC;
صعب تمرين 11: تقرير المدير - 4 جداول◀
المدير عايز تقرير فيه: اسم كل قسم، مكانه، عدد الموظفين، متوسط المرتب، عدد المشاريع، إجمالي ميزانية المشاريع. للأقسام اللي فيها موظفين بس.
departments INNER JOIN employees LEFT JOIN projects. استخدم DISTINCT في COUNT و SUM عشان التكرار. GROUP BY dept_name و location.
SELECT d.dept_name, d.location, COUNT(DISTINCT e.emp_id) AS emp_count, ROUND(AVG(DISTINCT e.salary)) AS avg_salary, COUNT(DISTINCT p.project_id) AS project_count, SUM(DISTINCT p.budget) AS total_project_budget FROM departments d INNER JOIN employees e ON d.dept_id = e.dept_id LEFT JOIN projects p ON d.dept_id = p.dept_id GROUP BY d.dept_name, d.location ORDER BY emp_count DESC;
التحدي النهائي تمرين 12: التقرير الشامل◀
اعرض تقرير شامل لكل موظف فيه:
- اسمه الكامل (first + last)
- مرتبه (لو NULL اعرض "لم يحدد")
- اسم قسمه (لو NULL اعرض "بدون قسم")
- مكان القسم
- اسم مديره (لو NULL اعرض "المدير العام")
- عدد العملاء المسؤول عنهم
رتب حسب عدد العملاء تنازلياً ثم المرتب تنازلياً.
4 JOINs: employees LEFT JOIN departments + LEFT JOIN employees (manager) + LEFT JOIN customers. GROUP BY كل الأعمدة اللي مش في دوال تجميعية. NVL للـ NULLs. TO_CHAR للمرتب.
SELECT e.first_name || ' ' || e.last_name AS full_name, NVL(TO_CHAR(e.salary), 'لم يحدد') AS salary, NVL(d.dept_name, 'بدون قسم') AS department, NVL(d.location, '-') AS location, NVL(m.first_name || ' ' || m.last_name, 'المدير العام') AS manager, COUNT(c.customer_id) AS customer_count FROM employees e LEFT JOIN departments d ON e.dept_id = d.dept_id LEFT JOIN employees m ON e.manager_id = m.emp_id LEFT JOIN customers c ON e.emp_id = c.assigned_emp_id GROUP BY e.first_name, e.last_name, e.salary, d.dept_name, d.location, m.first_name, m.last_name ORDER BY customer_count DESC, e.salary DESC NULLS LAST;
