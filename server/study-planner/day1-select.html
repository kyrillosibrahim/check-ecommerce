SELECT - الأمر الأساسي

SELECT = "هاتلي" - بيطلب بيانات من جدول أو أكتر. هو أكتر أمر SQL هتستخدمه في حياتك!
📐 الصيغة الأساسية
SQL Syntax
SELECT column1, column2, ... FROM table_name;

هات كل حاجة - SELECT *

SQL
-- هاتلي كل الأعمدة وكل الصفوف من جدول الموظفين SELECT * FROM employees; -- النجمة * معناها "كل الأعمدة" -- النتيجة: الـ 10 صفوف كاملين بكل الأعمدة
⚠️
تحذير: في الشغل الحقيقي متستخدمش SELECT * على جداول كبيرة! لو الجدول فيه مليون صف هيطلعهم كلهم وده هيبطئ كل حاجة. حدد الأعمدة اللي محتاجها بس.

الأسماء المستعارة (Aliases)

ممكن تغير اسم العمود اللي بيظهر في النتيجة باستخدام AS - مجرد اسم عرض مش بيغير الجدول نفسه.

SQL - Aliases
-- غيّر اسم العمود في النتيجة SELECT first_name AS "الاسم", salary AS "المرتب", dept AS "القسم" FROM employees; -- ممكن كمان من غير AS (اختصار) SELECT first_name "الاسم", salary "المرتب" FROM employees; -- لو الاسم المستعار كلمة واحدة إنجليزي ممكن من غير "" SELECT first_name AS name, salary AS pay FROM employees;
🔢

العمليات الحسابية في SELECT

SQL - حسابات
-- المرتب السنوي (المرتب × 12) SELECT first_name, salary, salary * 12 AS "المرتب السنوي" FROM employees; -- المرتب بعد زيادة 10% SELECT first_name, salary AS "المرتب الحالي", salary * 1.10 AS "بعد الزيادة", salary * 0.10 AS "قيمة الزيادة" FROM employees; -- المرتب بعد خصم الضرائب (14%) SELECT first_name, salary AS "قبل الضرائب", salary * 0.86 AS "بعد الضرائب" FROM employees;

WHERE - فلترة البيانات

WHERE = "بس بشرط" - بيفلتر النتايج ويرجعلك الصفوف اللي بتحقق الشرط بس
📐 الصيغة
SQL Syntax
SELECT columns FROM table_name WHERE condition;
SQL - أول WHERE
-- هاتلي الموظفين اللي في قسم IT بس SELECT first_name, salary, dept FROM employees WHERE dept = 'IT';

AND و OR و NOT - دمج الشروط

العاملالمعنىمثال
ANDالشرطين لازم يتحققوا مع بعضفي IT و مرتبه عالي
ORأي واحد من الشروط يكفيفي IT أو في HR
NOTعكس الشرطمش في مبيعات
SQL - AND, OR, NOT
-- AND: في IT ومرتبه أكبر من 12000 SELECT first_name, salary, dept FROM employees WHERE dept = 'IT' AND salary > 12000; -- النتيجة: نورا (15000) -- OR: في IT أو HR SELECT first_name, dept FROM employees WHERE dept = 'IT' OR dept = 'HR'; -- النتيجة: سارة، نورا، عمر، ياسمين، مريم -- NOT: مش في المبيعات SELECT first_name, dept FROM employees WHERE NOT dept = 'مبيعات'; -- دمج AND و OR (استخدم أقواس!) SELECT first_name, salary, dept FROM employees WHERE (dept = 'IT' OR dept = 'HR') AND salary > 9000; -- (في IT أو HR) و (مرتبه فوق 9000)
💡
نصيحة مهمة: لما تخلط AND مع OR استخدم أقواس ( ) دايماً عشان Oracle يفهم الأولوية صح. AND بينفذ قبل OR لو مفيش أقواس!

BETWEEN, IN, LIKE - عوامل خاصة

1. BETWEEN - بين قيمتين

SQL - BETWEEN
-- مرتب بين 7000 و 10000 (شامل الحدين) SELECT first_name, salary FROM employees WHERE salary BETWEEN 7000 AND 10000; -- نفس الحاجة بالظبط بس بصيغة تانية: -- WHERE salary >= 7000 AND salary <= 10000 -- اتعين في سنة 2023 SELECT first_name, hire_date FROM employees WHERE hire_date BETWEEN TO_DATE('2023-01-01','YYYY-MM-DD') AND TO_DATE('2023-12-31','YYYY-MM-DD');

2. IN - ضمن قائمة

SQL - IN
-- في أحد الأقسام دي SELECT first_name, dept FROM employees WHERE dept IN ('IT', 'HR', 'تسويق'); -- بدل ما تكتب: dept = 'IT' OR dept = 'HR' OR dept = 'تسويق' -- IN أقصر وأوضح ✅ -- مش في أي قسم من دول SELECT first_name, dept FROM employees WHERE dept NOT IN ('IT', 'HR');

3. LIKE - البحث بالنمط (Pattern)

🔤 رموز LIKE
الرمزالمعنىمثالبيطابق
%أي عدد من الحروف (حتى صفر)'أ%'أحمد، أمل، أ
_حرف واحد بالظبط'_مر'عمر (3 حروف بيبدأ بأي حرف)
SQL - LIKE
-- الاسم بيبدأ بحرف "م" SELECT first_name FROM employees WHERE first_name LIKE 'م%'; -- النتيجة: محمد، مريم -- الإيميل فيه "mail" SELECT email FROM employees WHERE email LIKE '%mail%'; -- الاسم 4 حروف بالظبط SELECT first_name FROM employees WHERE first_name LIKE '____'; -- 4 underscores = 4 حروف -- الاسم بينتهي بـ "ة" SELECT first_name FROM employees WHERE first_name LIKE '%ة'; -- النتيجة: فاطمة
❓

IS NULL / IS NOT NULL

NULL = "مفيش قيمة" - مش صفر ومش فاضي، يعني القيمة مجهولة أصلاً. مينفعش تقارنها بـ = لازم تستخدم IS NULL
SQL - NULL
-- ❌ غلط! ده مش هيشتغل: SELECT * FROM employees WHERE phone = NULL; -- ❌ WRONG -- ✅ صح! استخدم IS NULL: SELECT * FROM employees WHERE phone IS NULL; -- اللي عندهم تليفون SELECT * FROM employees WHERE phone IS NOT NULL;
🔃

ORDER BY - ترتيب النتائج

ORDER BY = "رتبهالي" - بيرتب النتايج تصاعدياً (ASC) أو تنازلياً (DESC). بدون ORDER BY الترتيب مش مضمون!
SQL - ORDER BY
-- رتب حسب المرتب تصاعدياً (من الأقل للأعلى) - الافتراضي SELECT first_name, salary FROM employees ORDER BY salary; -- أو salary ASC -- رتب حسب المرتب تنازلياً (من الأعلى للأقل) SELECT first_name, salary FROM employees ORDER BY salary DESC; -- الأعلى مرتب الأول

ترتيب بأكتر من عمود

SQL - ترتيب متعدد
-- رتب حسب القسم، وجوا كل قسم حسب المرتب تنازلياً SELECT first_name, dept, salary FROM employees ORDER BY dept ASC, salary DESC; -- الأقسام أبجدياً، وجوا كل قسم الأعلى مرتب الأول -- ترتيب حسب تاريخ التعيين (الأحدث الأول) SELECT first_name, hire_date FROM employees ORDER BY hire_date DESC; -- ترتيب حسب رقم العمود (مش الاسم) SELECT first_name, salary FROM employees ORDER BY 2 DESC; -- العمود التاني = salary

DISTINCT - إزالة التكرار

DISTINCT = "بدون تكرار" - بيشيل الصفوف المكررة من النتيجة ويرجع القيم الفريدة بس
SQL - بدون DISTINCT
-- بدون DISTINCT: هيظهر تكرار SELECT dept FROM employees; -- مبيعات، IT، مبيعات، IT، تسويق، HR، IT، مبيعات، تسويق، HR -- ↑ الأقسام متكررة! 10 صفوف
SQL - مع DISTINCT
-- مع DISTINCT: كل قسم مرة واحدة بس SELECT DISTINCT dept FROM employees; -- مبيعات، IT، تسويق، HR -- ↑ 4 صفوف بس! بدون تكرار ✅

SQL - أمثلة DISTINCT
-- المدن الموجودة SELECT DISTINCT city FROM employees; -- القاهرة، الإسكندرية، الجيزة، المنصورة -- DISTINCT على أكتر من عمود (التوليفة الفريدة) SELECT DISTINCT dept, city FROM employees; -- (مبيعات، القاهرة)، (مبيعات، المنصورة)، (IT، القاهرة)، ... -- كم مدينة عندنا؟ SELECT COUNT(DISTINCT city) AS "عدد المدن" FROM employees; -- النتيجة: 4

دمج كل حاجة مع بعض

SQL - أمثلة مركبة
-- 🔥 أعلى 3 مرتبات في IT SELECT first_name, salary FROM employees WHERE dept = 'IT' ORDER BY salary DESC FETCH FIRST 3 ROWS ONLY; -- 🔥 الموظفين في القاهرة بمرتب بين 7000 و 12000 مرتبين بالاسم SELECT first_name, salary, dept FROM employees WHERE city = 'القاهرة' AND salary BETWEEN 7000 AND 12000 ORDER BY first_name; -- 🔥 الأقسام الموجودة في الإسكندرية (بدون تكرار) SELECT DISTINCT dept FROM employees WHERE city = 'الإسكندرية'; -- 🔥 كل الموظفين ماعدا مبيعات، الأعلى مرتب الأول SELECT first_name || ' ' || last_name AS "الاسم", salary AS "المرتب", salary * 12 AS "السنوي" FROM employees WHERE dept NOT IN ('مبيعات') ORDER BY salary DESC;

ترتيب التنفيذ في Oracle

ده الترتيب اللي Oracle بينفذ بيه الأمر (مش الترتيب اللي بتكتبه!):

1️⃣ FROM
←
2️⃣ WHERE
←
3️⃣ SELECT
←
4️⃣ DISTINCT
←
5️⃣ ORDER BY
الترتيبالأمربيعمل إيه
1FROMبيحدد الجدول
2WHEREبيفلتر الصفوف
3SELECTبيختار الأعمدة
4DISTINCTبيشيل التكرار
5ORDER BYبيرتب النتيجة النهائية
💡
عشان كده مينفعش تستخدم Alias بتاع SELECT في WHERE! لأن WHERE بتتنفذ قبل SELECT. بس ممكن تستخدمه في ORDER BY لأنه بيتنفذ بعد SELECT.

SQL - الحلول

📌 ملخص الدرس
  • ✓ SELECT * = كل الأعمدة
  • ✓ SELECT col = أعمدة محددة
  • ✓ AS = اسم مستعار
  • ✓ || = دمج نصوص
  • ✓ WHERE = يساوي
  • ✓ WHERE > < أكبر / أصغر
  • ✓ AND = الشرطين مع بعض
  • ✓ OR = أي شرط يكفي
  • ✓ NOT = عكس الشرط
  • ✓ BETWEEN = بين قيمتين
  • ✓ IN = ضمن قائمة
  • ✓ LIKE % = بحث بنمط
  • ✓ IS NULL = قيمة فارغة
  • ✓ ORDER BY = ترتيب
  • ✓ DESC / ASC = تنازلي / تصاعدي
  • ✓ DISTINCT = بدون تكرار
