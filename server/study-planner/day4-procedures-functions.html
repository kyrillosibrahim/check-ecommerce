Procedures vs Functions

هتتعلم الفرق بين الـ Procedure والـ Function، وامتى تستخدم كل واحدة، وإزاي تستخدم الـ Function جوه SELECT!

المقدمة

إيه هي الـ Function؟

🧮

تشبيه: الآلة الحاسبة!

الـ Function زي الآلة الحاسبة: بتدخّل أرقام (inputs) وبترجع لك نتيجة (RETURN). دايماً بترجع قيمة واحدة. الـ Procedure بتنفذ عملية بس ممكن مترجعش حاجة.

📥

Input

بتاخد Parameters (صفر أو أكتر)

⚙️

Process

بتعمل حسابات أو عمليات

📤

RETURN

لازم ترجع قيمة واحدة!

الصيغة

صيغة CREATE FUNCTION

PL/SQL - Function Syntax
CREATE OR REPLACE FUNCTION function_name(
    parameter1 IN datatype,
    parameter2 IN datatype
)
RETURN return_datatype
IS
    -- متغيرات محلية
BEGIN
    -- الكود
    RETURN value;  -- لازم!
END;
/
🚨 الفرق عن Procedure: الـ Function لازم فيها RETURN datatype بعد الـ Parameters ولازم فيها RETURN value جوه الـ BEGIN.
أول Function

أول Function: حساب الضريبة

PL/SQL - Tax Function
CREATE OR REPLACE FUNCTION calc_tax(
    p_salary IN NUMBER
)
RETURN NUMBER
IS
    c_tax_rate CONSTANT NUMBER := 0.14;
BEGIN
    RETURN ROUND(p_salary * c_tax_rate, 2);
END;
/

-- استخدام في PL/SQL block:
BEGIN
    DBMS_OUTPUT.PUT_LINE('الضريبة: ' || calc_tax(8000));  -- 1120
END;
/

-- استخدام في SQL مباشرة!
SELECT first_name, salary, calc_tax(salary) AS tax
FROM employees;
✅ شفت الفرق؟ الـ Function تقدر تستخدمها في SELECT مباشرة! الـ Procedure مش تقدر.
في SQL

Function جوه SELECT Statement

PL/SQL - Functions
-- Function بتجيب اسم القسم
CREATE OR REPLACE FUNCTION get_dept_name(
    p_dept_id IN NUMBER
)
RETURN VARCHAR2
IS
    v_name departments.department_name%TYPE;
BEGIN
    SELECT department_name INTO v_name
    FROM departments WHERE department_id = p_dept_id;
    RETURN v_name;
EXCEPTION
    WHEN NO_DATA_FOUND THEN RETURN 'غير معروف';
END;
/

-- استخدمها في SELECT بدل JOIN!
SELECT
    employee_id,
    first_name,
    salary,
    get_dept_name(department_id) AS dept_name
FROM employees
WHERE salary > 5000;
PL/SQL - Function لحساب صافي المرتب
CREATE OR REPLACE FUNCTION calc_net_salary(
    p_salary IN NUMBER,
    p_comm   IN NUMBER DEFAULT 0
)
RETURN NUMBER
IS
BEGIN
    RETURN ROUND(p_salary + NVL(p_salary * p_comm, 0) - (p_salary * 0.14), 2);
END;
/

SELECT first_name, salary,
    calc_net_salary(salary, commission_pct) AS net_salary
FROM employees ORDER BY net_salary DESC;
المقارنة الكبيرة

Procedure vs Function

📋 Procedure

  • بتنفذ عملية (action)
  • مش لازم ترجع قيمة
  • ممكن OUT parameters
  • بتستدعيها بـ EXEC أو في BEGIN/END
  • مش تقدر تستخدمها في SELECT
  • ممكن تعمل DML (INSERT/UPDATE/DELETE)
VS

🧮 Function

  • بترجع قيمة (value)
  • لازم RETURN
  • عادةً IN parameters بس
  • بتستدعيها في أي expression
  • تقدر تستخدمها في SELECT!
  • الأفضل تكون للقراءة بس (no DML)

جدول المقارنة:

الخاصيةProcedureFunction
RETURN clause❌ لا✅ إجباري
OUT parameters✅ نعمممكن بس مش مستحب
في SELECT❌ لا✅ نعم!
في WHERE❌ لا✅ نعم!
في PL/SQL expression❌ لا✅ نعم
DML (INSERT/UPDATE)✅ طبيعيممكن بس مش مستحب
الهدفتنفيذ عمليةحساب وإرجاع قيمة
أنواع

Functions بأنواع رجوع مختلفة

Function بترجع VARCHAR2

PL/SQL
CREATE OR REPLACE FUNCTION get_salary_level(
    p_salary IN NUMBER
)
RETURN VARCHAR2
IS
BEGIN
    CASE
        WHEN p_salary >= 15000 THEN RETURN 'Senior';
        WHEN p_salary >= 8000  THEN RETURN 'Mid-Level';
        WHEN p_salary >= 4000  THEN RETURN 'Junior';
        ELSE RETURN 'Trainee';
    END CASE;
END;
/

SELECT first_name, salary, get_salary_level(salary) AS level
FROM employees;

Function بترجع BOOLEAN

PL/SQL
-- BOOLEAN Functions بتشتغل في PL/SQL بس (مش في SQL!)
CREATE OR REPLACE FUNCTION is_manager(
    p_emp_id IN NUMBER
)
RETURN BOOLEAN
IS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count
    FROM employees WHERE manager_id = p_emp_id;
    RETURN v_count > 0;
END;
/

BEGIN
    IF is_manager(100) THEN
        DBMS_OUTPUT.PUT_LINE('ده مدير!');
    END IF;
END;
/
⚠️ BOOLEAN Function مش تقدر تستخدمها في SELECT! لأن SQL مش بيدعم BOOLEAN. لو عايزها في SELECT، خليها ترجع NUMBER (1/0) أو VARCHAR2 ('Y'/'N').
متقدم

Function مع Parameters متعددة و DEFAULT

PL/SQL - Multiple Parameters
CREATE OR REPLACE FUNCTION calc_bonus(
    p_salary     IN NUMBER,
    p_years      IN NUMBER,
    p_base_rate  IN NUMBER DEFAULT 5  -- 5% default
)
RETURN NUMBER
IS
    v_rate NUMBER;
BEGIN
    -- كل سنة خبرة بتزود 1%
    v_rate := p_base_rate + p_years;
    RETURN ROUND(p_salary * v_rate / 100, 2);
END;
/

-- استخدام:
SELECT first_name, salary,
    ROUND(MONTHS_BETWEEN(SYSDATE,hire_date)/12) AS years,
    calc_bonus(salary, ROUND(MONTHS_BETWEEN(SYSDATE,hire_date)/12)) AS bonus
FROM employees;
الاستخدام

كل الأماكن اللي تقدر تستخدم فيها Function

SQL - Function في كل مكان!
-- 1. في SELECT
SELECT first_name, calc_tax(salary) FROM employees;

-- 2. في WHERE
SELECT * FROM employees
WHERE get_salary_level(salary) = 'Senior';

-- 3. في ORDER BY
SELECT * FROM employees
ORDER BY calc_net_salary(salary, commission_pct) DESC;

-- 4. في PL/SQL variable assignment
v_tax := calc_tax(8000);

-- 5. في DBMS_OUTPUT
DBMS_OUTPUT.PUT_LINE('الضريبة: ' || calc_tax(8000));

-- 6. كـ parameter لـ function تانية!
v_net := p_salary - calc_tax(p_salary);

القاعدة الذهبية:

الحالةاستخدممثال

مقدمة

مقدمة Packages

الـ Package = مجموعة من Procedures و Functions مرتبطة ببعض محطوطة في وحدة واحدة.

PL/SQL - Package
-- 1. Package Specification (الواجهة)
CREATE OR REPLACE PACKAGE hr_pkg AS
    FUNCTION calc_tax(p_salary NUMBER) RETURN NUMBER;
    FUNCTION get_level(p_salary NUMBER) RETURN VARCHAR2;
    PROCEDURE raise_salary(p_emp_id NUMBER, p_pct NUMBER);
END hr_pkg;
/

-- 2. Package Body (التنفيذ)
CREATE OR REPLACE PACKAGE BODY hr_pkg AS
    FUNCTION calc_tax(p_salary NUMBER) RETURN NUMBER IS
    BEGIN
        RETURN ROUND(p_salary * 0.14, 2);
    END;

    FUNCTION get_level(p_salary NUMBER) RETURN VARCHAR2 IS
    BEGIN
        IF p_salary >= 15000 THEN RETURN 'Senior';
        ELSIF p_salary >= 8000 THEN RETURN 'Mid';
        ELSE RETURN 'Junior';
        END IF;
    END;

    PROCEDURE raise_salary(p_emp_id NUMBER, p_pct NUMBER) IS
    BEGIN
        UPDATE employees SET salary = salary * (1 + p_pct/100)
        WHERE employee_id = p_emp_id;
        COMMIT;
    END;
END hr_pkg;
/

-- استخدام:
SELECT first_name, hr_pkg.calc_tax(salary) FROM employees;
EXEC hr_pkg.raise_salary(100, 10);
✅ فوائد Packages: تنظيم الكود + إخفاء التفاصيل + أداء أفضل (بيتحمل مرة واحدة في الذاكرة) + تقدر تعمل variables مشتركة بين كل الـ procedures/functions.
تمارين

تمارين عملية

تمرين 1: Function بسيطة

سهل

اكتب Function اسمها double_salary بتاخد مرتب وبترجع الضعف.

PL/SQL
CREATE OR REPLACE FUNCTION double_salary(p_salary IN NUMBER)
RETURN NUMBER IS
BEGIN
    RETURN p_salary * 2;
END;
/
SELECT first_name, salary, double_salary(salary) AS doubled FROM employees;

تمرين 2: Function بترجع نص

سهل

اكتب Function اسمها full_name بتاخد first_name و last_name وبترجعهم مع بعض بمسافة.

PL/SQL
CREATE OR REPLACE FUNCTION full_name(
    p_first IN VARCHAR2, p_last IN VARCHAR2
) RETURN VARCHAR2 IS
BEGIN
    RETURN p_first || ' ' || p_last;
END;
/
SELECT full_name(first_name, last_name) AS name, salary FROM employees;

تمرين 3: Function مع IF

متوسط

اكتب Function اسمها get_grade بتاخد درجة وبترجع التقدير (A/B/C/D/F).

PL/SQL
CREATE OR REPLACE FUNCTION get_grade(p_score IN NUMBER)
RETURN VARCHAR2 IS
BEGIN
    IF p_score >= 90 THEN RETURN 'A';
    ELSIF p_score >= 80 THEN RETURN 'B';
    ELSIF p_score >= 70 THEN RETURN 'C';
    ELSIF p_score >= 60 THEN RETURN 'D';
    ELSE RETURN 'F';
    END IF;
END;
/

تمرين 4: Function بتجيب بيانات

متوسط

اكتب Function اسمها get_emp_salary بتاخد employee_id وبترجع مرتبه. لو مش موجود ترجع -1.

PL/SQL
CREATE OR REPLACE FUNCTION get_emp_salary(p_id IN NUMBER)
RETURN NUMBER IS
    v_sal employees.salary%TYPE;
BEGIN
    SELECT salary INTO v_sal FROM employees WHERE employee_id = p_id;
    RETURN v_sal;
EXCEPTION
    WHEN NO_DATA_FOUND THEN RETURN -1;
END;
/

تمرين 5: Function vs Procedure

متوسط

اكتب Function count_dept_emps بتاخد dept_id وبترجع عدد الموظفين. ثم استخدمها في SELECT عشان تعرض اسم كل قسم وعدد موظفينه.

PL/SQL
CREATE OR REPLACE FUNCTION count_dept_emps(p_dept IN NUMBER)
RETURN NUMBER IS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count FROM employees WHERE department_id = p_dept;
    RETURN v_count;
END;
/

SELECT department_id, department_name,
    count_dept_emps(department_id) AS emp_count
FROM departments
ORDER BY emp_count DESC;

تمرين 6: Function حساب سنين الخبرة

صعب

اكتب Function years_of_service بتاخد hire_date وبترجع عدد السنين. استخدمها في SELECT مع كل الموظفين.

PL/SQL
CREATE OR REPLACE FUNCTION years_of_service(p_hire IN DATE)
RETURN NUMBER IS
BEGIN
    RETURN ROUND(MONTHS_BETWEEN(SYSDATE, p_hire) / 12);
END;
/

SELECT first_name, hire_date, years_of_service(hire_date) AS years
FROM employees ORDER BY years DESC;

تمرين 7: Procedure مع Function جواها

صعب

اكتب Procedure print_emp_report بتستخدم جواها Functions: get_dept_name + calc_tax + get_salary_level عشان تطبع تقرير كامل لموظف.

PL/SQL
CREATE OR REPLACE PROCEDURE print_emp_report(p_id IN NUMBER)
IS
    v_emp employees%ROWTYPE;
BEGIN
    SELECT * INTO v_emp FROM employees WHERE employee_id = p_id;

    DBMS_OUTPUT.PUT_LINE('=== تقرير موظف ===');
    DBMS_OUTPUT.PUT_LINE('الاسم: ' || full_name(v_emp.first_name, v_emp.last_name));
    DBMS_OUTPUT.PUT_LINE('القسم: ' || get_dept_name(v_emp.department_id));
    DBMS_OUTPUT.PUT_LINE('المستوى: ' || get_salary_level(v_emp.salary));
    DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_emp.salary);
    DBMS_OUTPUT.PUT_LINE('الضريبة: ' || calc_tax(v_emp.salary));
    DBMS_OUTPUT.PUT_LINE('الصافي: ' || (v_emp.salary - calc_tax(v_emp.salary)));
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('الموظف مش موجود!');
END;
/
EXEC print_emp_report(100);

تمرين 8: Package صغير

تحدي

اعمل Package اسمه salary_pkg فيه: Function calc_tax، Function calc_net، Procedure print_payslip بتطبع كل التفاصيل.

PL/SQL
CREATE OR REPLACE PACKAGE salary_pkg AS
    FUNCTION calc_tax(p_sal NUMBER) RETURN NUMBER;
    FUNCTION calc_net(p_sal NUMBER) RETURN NUMBER;
    PROCEDURE print_payslip(p_emp_id NUMBER);
END;
/

CREATE OR REPLACE PACKAGE BODY salary_pkg AS
    FUNCTION calc_tax(p_sal NUMBER) RETURN NUMBER IS
    BEGIN RETURN ROUND(p_sal * 0.14, 2); END;

    FUNCTION calc_net(p_sal NUMBER) RETURN NUMBER IS
    BEGIN RETURN p_sal - calc_tax(p_sal); END;

    PROCEDURE print_payslip(p_emp_id NUMBER) IS
        v_emp employees%ROWTYPE;
    BEGIN
        SELECT * INTO v_emp FROM employees WHERE employee_id = p_emp_id;
        DBMS_OUTPUT.PUT_LINE('الاسم: ' || v_emp.first_name);
        DBMS_OUTPUT.PUT_LINE('المرتب: ' || v_emp.salary);
        DBMS_OUTPUT.PUT_LINE('الضريبة: ' || calc_tax(v_emp.salary));
        DBMS_OUTPUT.PUT_LINE('الصافي: ' || calc_net(v_emp.salary));
    END;
END;
/
EXEC salary_pkg.print_payslip(100);

ملخص اللي اتعلمناه

  • Function لازم فيها RETURN datatype و RETURN value
  • Function تقدر تستخدمها في SELECT و WHERE و ORDER BY
  • Procedure للـ actions (INSERT/UPDATE/DELETE)، Function للحسابات وإرجاع القيم
  • Function في PL/SQL بتشتغل زي أي built-in function (مثل ROUND, UPPER)
  • BOOLEAN Function تشتغل في PL/SQL بس، مش في SQL
  • Package = مجموعة Procedures + Functions في وحدة واحدة
  • Package ليه Specification (واجهة) و Body (تنفيذ)
  • القاعدة: "إيه القيمة؟" → Function | "اعمل ده" → Procedure
