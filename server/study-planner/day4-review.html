Aggregate Functions + GROUP BY + HAVING

SQL - GROUP BY
SELECT department_id,
       COUNT(*) AS emp_count,
       AVG(salary) AS avg_sal,
       MAX(salary) AS max_sal,
       MIN(salary) AS min_sal,
       SUM(salary) AS total_sal
FROM employees
GROUP BY department_id
HAVING COUNT(*) > 5
ORDER BY avg_sal DESC;

تمرين 1: SELECT مع WHERE

سهل

اكتب استعلام يجيب أسماء وأول حرف من اسم العائلة ومرتبات كل الموظفين اللي مرتبهم أكتر من 7000 ومن قسم 30 أو 40، مرتبين بالمرتب تنازلياً.

تلميح: استخدم WHERE مع AND و IN أو OR. استخدم SUBSTR للحرف الأول. رتّب بـ ORDER BY DESC.
SQL
SELECT first_name,
       SUBSTR(last_name, 1, 1) AS last_initial,
       salary
FROM employees
WHERE salary > 7000
  AND department_id IN (30, 40)
ORDER BY salary DESC;

تمرين 2: JOIN بين جدولين

سهل

اكتب استعلام يعرض اسم الموظف واسم القسم بتاعه لكل الموظفين، بما فيهم الأقسام اللي مفيهاش موظفين.

تلميح: استخدم RIGHT JOIN عشان تجيب كل الأقسام حتى لو مفيش فيها موظفين. أو LEFT JOIN بالعكس.
SQL
SELECT e.first_name,
       e.last_name,
       d.department_name
FROM employees e
RIGHT JOIN departments d
  ON e.department_id = d.department_id
ORDER BY d.department_name;

تمرين 3: GROUP BY مع HAVING

سهل

اكتب استعلام يعرض اسم كل قسم وعدد الموظفين فيه ومتوسط المرتب، بس للأقسام اللي فيها أكتر من موظف واحد. رتّب حسب عدد الموظفين تنازلياً.

تلميح: محتاج JOIN بين employees و departments، ثم GROUP BY اسم القسم، ثم HAVING COUNT(*) > 1.
SQL
SELECT d.department_name,
       COUNT(*) AS emp_count,
       ROUND(AVG(e.salary), 2) AS avg_salary
FROM employees e
JOIN departments d
  ON e.department_id = d.department_id
GROUP BY d.department_name
HAVING COUNT(*) > 1
ORDER BY emp_count DESC;

تمرين 4: Subquery

متوسط

اكتب استعلام يجيب الموظفين اللي مرتبهم أعلى من متوسط مرتب القسم بتاعهم. اعرض الاسم والمرتب واسم القسم.

تلميح: استخدم Correlated Subquery في WHERE: salary > (SELECT AVG(salary) FROM employees WHERE department_id = e.department_id)
SQL
SELECT e.first_name,
       e.salary,
       d.department_name,
       (SELECT ROUND(AVG(salary), 2)
        FROM employees
        WHERE department_id = e.department_id) AS dept_avg
FROM employees e
JOIN departments d ON e.department_id = d.department_id
WHERE e.salary > (
    SELECT AVG(salary)
    FROM employees
    WHERE department_id = e.department_id
)
ORDER BY e.salary DESC;

تمرين 5: إنشاء View

متوسط

اعمل View اسمها v_dept_summary بتعرض: اسم القسم، عدد الموظفين، إجمالي المرتبات، أعلى مرتب، وأقل مرتب. بعد كده استعلم منها عن الأقسام اللي إجمالي مرتباتها أكتر من 15000.

تلميح: استخدم CREATE VIEW مع JOIN و GROUP BY. بعدين SELECT من الـ View مع WHERE.
SQL
CREATE OR REPLACE VIEW v_dept_summary AS
SELECT d.department_name,
       COUNT(e.employee_id) AS emp_count,
       SUM(e.salary) AS total_salary,
       MAX(e.salary) AS max_salary,
       MIN(e.salary) AS min_salary
FROM departments d
LEFT JOIN employees e ON d.department_id = e.department_id
GROUP BY d.department_name;

-- استعلام من الـ View
SELECT * FROM v_dept_summary
WHERE total_salary > 15000
ORDER BY total_salary DESC;

تمرين 6: Aggregate Functions متقدمة

متوسط

اكتب استعلام يعرض لكل موظف: اسمه، مرتبه، الفرق بين مرتبه وأعلى مرتب في الشركة كلها، والنسبة المئوية لمرتبه من إجمالي مرتبات الشركة (مقربة لخانتين).

تلميح: استخدم Subquery في SELECT: (SELECT MAX(salary) FROM employees) - salary للفرق. واستخدم ROUND(salary / (SELECT SUM(salary) FROM employees) * 100, 2) للنسبة.
SQL
SELECT first_name,
       salary,
       (SELECT MAX(salary) FROM employees) - salary AS diff_from_max,
       ROUND(salary / (SELECT SUM(salary) FROM employees) * 100, 2) AS salary_pct
FROM employees
ORDER BY salary DESC;

تمرين 7: PL/SQL - Cursor مع IF

صعب

اكتب بلوك PL/SQL يستخدم Cursor FOR LOOP يقرأ كل الموظفين ويصنفهم حسب المرتب:
- أكتر من 10000: "مستوى A - ممتاز"
- بين 5000 و 10000: "مستوى B - جيد"
- أقل من 5000: "مستوى C - يحتاج مراجعة"
واطبع الاسم والمرتب والمستوى واسم القسم. في الآخر اطبع عدد كل مستوى.

تلميح: عرّف 3 عدادات (v_count_a, v_count_b, v_count_c). استخدم JOIN في الـ Cursor عشان تجيب اسم القسم. استخدم IF/ELSIF/ELSE للتصنيف.
PL/SQL
DECLARE
    v_level   VARCHAR2(30);
    v_count_a NUMBER := 0;
    v_count_b NUMBER := 0;
    v_count_c NUMBER := 0;
BEGIN
    DBMS_OUTPUT.PUT_LINE('===== تقرير تصنيف الموظفين =====');
    DBMS_OUTPUT.PUT_LINE('--------------------------------');

    FOR emp IN (
        SELECT e.first_name, e.salary, d.department_name
        FROM employees e
        JOIN departments d ON e.department_id = d.department_id
        ORDER BY e.salary DESC
    ) LOOP
        IF emp.salary > 10000 THEN
            v_level := 'مستوى A - ممتاز';
            v_count_a := v_count_a + 1;
        ELSIF emp.salary >= 5000 THEN
            v_level := 'مستوى B - جيد';
            v_count_b := v_count_b + 1;
        ELSE
            v_level := 'مستوى C - يحتاج مراجعة';
            v_count_c := v_count_c + 1;
        END IF;

        DBMS_OUTPUT.PUT_LINE(
            emp.first_name || ' | ' ||
            emp.salary || ' | ' ||
            emp.department_name || ' | ' ||
            v_level
        );
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('================================');
    DBMS_OUTPUT.PUT_LINE('مستوى A (ممتاز): ' || v_count_a);
    DBMS_OUTPUT.PUT_LINE('مستوى B (جيد): ' || v_count_b);
    DBMS_OUTPUT.PUT_LINE('مستوى C (مراجعة): ' || v_count_c);
END;
/

تمرين 8: PL/SQL - Exception Handling

صعب

اكتب بلوك PL/SQL يجيب اسم ومرتب موظف معين بالـ employee_id. لو الموظف مش موجود اطبع رسالة خطأ. لو المرتب أقل من 5000 طلّق User-Defined Exception اسمه e_low_salary واطبع رسالة "المرتب أقل من الحد الأدنى!". لو أي خطأ تاني اطبع الرسالة بـ SQLERRM.

تلميح: عرّف e_low_salary EXCEPTION في DECLARE. بعد SELECT INTO فحص المرتب بـ IF، ولو أقل من 5000 اعمل RAISE e_low_salary. اتعامل مع NO_DATA_FOUND و e_low_salary و OTHERS.
PL/SQL
DECLARE
    v_name   employees.first_name%TYPE;
    v_salary employees.salary%TYPE;
    v_emp_id NUMBER := 110;  -- غيّر الرقم للتجربة
    e_low_salary EXCEPTION;
BEGIN
    SELECT first_name, salary
    INTO v_name, v_salary
    FROM employees
    WHERE employee_id = v_emp_id;

    DBMS_OUTPUT.PUT_LINE('الموظف: ' || v_name || ' - المرتب: ' || v_salary);

    IF v_salary < 5000 THEN
        RAISE e_low_salary;
    ELSE
        DBMS_OUTPUT.PUT_LINE('المرتب مقبول.');
    END IF;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('خطأ: الموظف رقم ' || v_emp_id || ' مش موجود!');
    WHEN e_low_salary THEN
        DBMS_OUTPUT.PUT_LINE('تحذير: المرتب ' || v_salary || ' أقل من الحد الأدنى (5000)!');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('خطأ غير متوقع: ' || SQLERRM);
END;
/

تمرين 9: PL/SQL - FOR LOOP مع حسابات

صعب

اكتب بلوك PL/SQL يحسب إجمالي الزيادات السنوية لكل موظف. الزيادة هي:
- لو عدد سنين الخبرة أكتر من 5: زيادة 10%
- لو بين 3 و 5: زيادة 7%
- لو أقل من 3: زيادة 5%
اطبع لكل موظف: الاسم، المرتب الحالي، عدد السنين، نسبة الزيادة، والمرتب الجديد. في الآخر اطبع إجمالي الزيادات على مستوى الشركة.

تلميح: استخدم TRUNC(MONTHS_BETWEEN(SYSDATE, hire_date)/12) لحساب عدد السنين. استخدم Cursor FOR LOOP ومتغير v_total_raise لتجميع كل الزيادات.
PL/SQL
DECLARE
    v_years      NUMBER;
    v_pct        NUMBER;
    v_new_salary NUMBER;
    v_raise      NUMBER;
    v_total      NUMBER := 0;
BEGIN
    DBMS_OUTPUT.PUT_LINE('===== تقرير الزيادات السنوية =====');
    DBMS_OUTPUT.PUT_LINE('---------------------------------');

    FOR emp IN (
        SELECT first_name, salary, hire_date
        FROM employees
        ORDER BY hire_date
    ) LOOP
        v_years := TRUNC(MONTHS_BETWEEN(SYSDATE, emp.hire_date) / 12);

        IF v_years > 5 THEN
            v_pct := 10;
        ELSIF v_years >= 3 THEN
            v_pct := 7;
        ELSE
            v_pct := 5;
        END IF;

        v_raise := ROUND(emp.salary * v_pct / 100, 2);
        v_new_salary := emp.salary + v_raise;
        v_total := v_total + v_raise;

        DBMS_OUTPUT.PUT_LINE(
            emp.first_name || ' | مرتب: ' || emp.salary ||
            ' | سنين: ' || v_years ||
            ' | زيادة: ' || v_pct || '%' ||
            ' | جديد: ' || v_new_salary
        );
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('=================================');
    DBMS_OUTPUT.PUT_LINE('إجمالي الزيادات: ' || v_total);
END;
/

تمرين 10: Procedure كاملة

تحدي

اعمل Procedure اسمها raise_salary بتاخد employee_id ونسبة الزيادة. الـ Procedure لازم:
- تفحص الموظف موجود
- تفحص إن النسبة بين 1 و 50 (مش سالبة ولا مبالغ فيها)
- تفحص إن المرتب الجديد مش أكتر من 50000
- تطبع المرتب القديم والجديد
- تستخدم RAISE_APPLICATION_ERROR لكل فحص فاشل

تلميح: ابدأ بفحص النسبة (IF p_pct NOT BETWEEN 1 AND 50). بعدين SELECT INTO المرتب. بعدين احسب المرتب الجديد وفحصه. لو كله تمام UPDATE.
PL/SQL
CREATE OR REPLACE PROCEDURE raise_salary(
    p_emp_id IN NUMBER,
    p_pct    IN NUMBER
)
IS
    v_name    employees.first_name%TYPE;
    v_old_sal employees.salary%TYPE;
    v_new_sal NUMBER;
BEGIN
    -- فحص النسبة
    IF p_pct NOT BETWEEN 1 AND 50 THEN
        RAISE_APPLICATION_ERROR(-20001,
            'النسبة لازم تكون بين 1 و 50! أنت بعتت: ' || p_pct);
    END IF;

    -- جيب بيانات الموظف
    BEGIN
        SELECT first_name, salary
        INTO v_name, v_old_sal
        FROM employees
        WHERE employee_id = p_emp_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20002,
                'الموظف رقم ' || p_emp_id || ' مش موجود!');
    END;

    -- احسب المرتب الجديد
    v_new_sal := ROUND(v_old_sal * (1 + p_pct / 100), 2);

    -- فحص الحد الأقصى
    IF v_new_sal > 50000 THEN
        RAISE_APPLICATION_ERROR(-20003,
            'المرتب الجديد (' || v_new_sal || ') يتعدى الحد الأقصى 50000!');
    END IF;

    -- تنفيذ الزيادة
    UPDATE employees
    SET salary = v_new_sal
    WHERE employee_id = p_emp_id;

    DBMS_OUTPUT.PUT_LINE('تم زيادة مرتب ' || v_name);
    DBMS_OUTPUT.PUT_LINE('من ' || v_old_sal || ' إلى ' || v_new_sal);
    DBMS_OUTPUT.PUT_LINE('نسبة الزيادة: ' || p_pct || '%');
    COMMIT;
END;
/

-- تجربة
EXEC raise_salary(107, 15);

-- EXEC raise_salary(108, 300);  -- خطأ: نسبة غلط

تمرين 11: Function تُستخدم في SELECT

تحدي

اعمل Function اسمها get_tax بتاخد المرتب وبترجع قيمة الضريبة:
- مرتب أكتر من 10000: ضريبة 15%
- بين 5000 و 10000: ضريبة 10%
- أقل من 5000: ضريبة 5%
بعد كده استخدمها في SELECT عشان تعرض لكل موظف: اسمه، مرتبه، الضريبة، والصافي (المرتب - الضريبة).

تلميح: الـ Function بتاخد p_salary IN NUMBER وبترجع RETURN NUMBER. جواها IF/ELSIF/ELSE. بعدين SELECT first_name, salary, get_tax(salary), salary - get_tax(salary).
PL/SQL
CREATE OR REPLACE FUNCTION get_tax(
    p_salary IN NUMBER
)
RETURN NUMBER
IS
    v_tax NUMBER;
BEGIN
    IF p_salary > 10000 THEN
        v_tax := ROUND(p_salary * 0.15, 2);
    ELSIF p_salary >= 5000 THEN
        v_tax := ROUND(p_salary * 0.10, 2);
    ELSE
        v_tax := ROUND(p_salary * 0.05, 2);
    END IF;

    RETURN v_tax;
END;
/

-- استخدام الـ Function في SELECT
SELECT first_name,
       salary,
       get_tax(salary) AS tax_amount,
       salary - get_tax(salary) AS net_salary
FROM employees
ORDER BY salary DESC;

تمرين 12: Procedure مع Cursor + Exception

تحدي

اعمل Procedure اسمها dept_bonus_report بتاخد department_id وبتعمل:
- تفحص القسم موجود (Exception لو لأ)
- تستخدم Cursor تقرأ كل موظفين القسم
- لكل موظف: لو عنده commission_pct تحسب البونص (salary * commission_pct)، غير كده البونص = 0
- تطبع اسم كل موظف ومرتبه والبونص
- في الآخر تطبع إجمالي البونصات وعدد الموظفين اللي أخذوا بونص
- لو القسم فاضي تطبع رسالة "مفيش موظفين!"

تلميح: ابدأ بـ SELECT department_name INTO (للتأكد القسم موجود). استخدم Cursor FOR LOOP. استخدم NVL(commission_pct, 0) أو IF IS NOT NULL. عدّاد لعدد اللي أخذوا بونص ومتغير لإجمالي البونص.
PL/SQL
CREATE OR REPLACE PROCEDURE dept_bonus_report(
    p_dept_id IN NUMBER
)
IS
    v_dept_name  departments.department_name%TYPE;
    v_bonus      NUMBER;
    v_total      NUMBER := 0;
    v_count      NUMBER := 0;
    v_bonus_cnt  NUMBER := 0;
BEGIN
    -- فحص القسم
    BEGIN
        SELECT department_name INTO v_dept_name
        FROM departments
        WHERE department_id = p_dept_id;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            RAISE_APPLICATION_ERROR(-20001,
                'القسم رقم ' || p_dept_id || ' غير موجود!');
    END;

    DBMS_OUTPUT.PUT_LINE('╔══════════════════════════════════╗');
    DBMS_OUTPUT.PUT_LINE('  تقرير بونص قسم: ' || v_dept_name);
    DBMS_OUTPUT.PUT_LINE('╠══════════════════════════════════╣');

    -- قراءة الموظفين
    FOR emp IN (
        SELECT first_name, salary, commission_pct
        FROM employees
        WHERE department_id = p_dept_id
        ORDER BY salary DESC
    ) LOOP
        v_count := v_count + 1;

        IF emp.commission_pct IS NOT NULL THEN
            v_bonus := ROUND(emp.salary * emp.commission_pct, 2);
            v_bonus_cnt := v_bonus_cnt + 1;
        ELSE
            v_bonus := 0;
        END IF;

        v_total := v_total + v_bonus;

        DBMS_OUTPUT.PUT_LINE(
            emp.first_name || ' | مرتب: ' || emp.salary ||
            ' | بونص: ' || v_bonus
        );
    END LOOP;

    IF v_count = 0 THEN
        DBMS_OUTPUT.PUT_LINE('  مفيش موظفين في القسم ده!');
    ELSE
        DBMS_OUTPUT.PUT_LINE('╠══════════════════════════════════╣');
        DBMS_OUTPUT.PUT_LINE('  عدد الموظفين: ' || v_count);
        DBMS_OUTPUT.PUT_LINE('  أخذوا بونص: ' || v_bonus_cnt);
        DBMS_OUTPUT.PUT_LINE('  إجمالي البونصات: ' || v_total);
    END IF;

    DBMS_OUTPUT.PUT_LINE('╚══════════════════════════════════╝');
END;
/

-- تجربة
EXEC dept_bonus_report(30);  -- قسم فيه commission
EXEC dept_bonus_report(40);  -- قسم بدون commission
-- EXEC dept_bonus_report(99);  -- قسم مش موجود
